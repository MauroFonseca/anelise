\section{Multimídia e QoS}

\begin{frame}{Multimídia e qualidade de serviço}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Aplicações de multimídia}
      \begin{itemize}
        \item áudio e vídeo de rede (``mídia contínua'')
        \item requerem atraso e perda controlados
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{QoS}
      \begin{itemize}
        \item a rede oferece à aplicação o \textbf{nível de desempenho necessário} para que funcione
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.3cm}
  \begin{alertblock}{Problema}
    TCP/UDP/IP: ``serviço de melhor esforço'' --- \textbf{sem garantia sobre atraso e perda}.
  \end{alertblock}
  \begin{block}{Estratégias}
    \begin{itemize}
      \item aplicações usam técnicas de camada de aplicação para aliviar atraso/perda (buffering, adaptação de taxa, FEC)
      \item a rede deve evoluir para dar melhor suporte (classes de serviço)
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Como deve evoluir a Internet para multimídia?}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Filosofia de serviços integrados (Intserv)}
      \begin{itemize}
        \item mudanças fundamentais na Internet
        \item as aplicações \textbf{reservam largura de banda fim-a-fim}
        \item requer software novo e complexo em hosts e roteadores
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Filosofia \emph{laissez-faire}}
      \begin{itemize}
        \item sem mudanças importantes
        \item mais largura de banda quando for necessário
        \item distribuição de conteúdo, multicast na camada de aplicação
      \end{itemize}
      \textbf{Filosofia de serviços diferenciados (Diffserv)}
      \begin{itemize}
        \item menos mudanças na infraestrutura
        \item oferece serviço de 1ª e 2ª classe
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Classes de serviço e princípios}

\begin{frame}{Proporcionar múltiplas classes de serviço}
  \begin{itemize}
    \item até agora: fazer o melhor com o serviço de ``melhor esforço'' (uma única classe)
    \item alternativa: \textbf{múltiplas classes de serviço}
      \begin{itemize}
        \item particionar o tráfego em classes
        \item a rede trata classes diferentes de forma diferente (analogia: serviço VIP vs.\ normal)
      \end{itemize}
    \item granularidade: serviço \textbf{diferencial entre classes}, não entre conexões individuais
    \item história: bits de \textbf{ToS} (IPv4) / \textbf{Flow Label} (IPv6)
  \end{itemize}

\begin{frame}{Princípio 1: marcação de pacote}
  \begin{exampleblock}{Cenário}
    Telefone IP a 1~Mbps e FTP compartilham enlace de 1,5~Mbps.
    \begin{itemize}
      \item os surtos de FTP podem congestionar o roteador e causar perda de áudio
      \item deseja-se dar \textbf{prioridade ao áudio} sobre o FTP
    \end{itemize}
  \end{exampleblock}
  \begin{block}{Princípio 1}
    \textbf{Marcação de pacote} necessária para que o roteador distinga entre classes; e nova \textbf{política de encaminhamento} para tratar os pacotes.
  \end{block}
\end{frame}

\begin{frame}{Princípio 2: marcação e shaping na borda}
  \begin{itemize}
    \item e se as aplicações se comportam mal (áudio envia mais do que o declarado)?
    \item \textbf{Shaping}: forçar a adesão da origem às alocações de largura de banda
  \end{itemize}

  \end{center}
  \begin{block}{Princípio 2}
    Marcação e shaping de pacote: \textbf{fornecer proteção (isolamento)} de uma classe frente a outras.
  \end{block}
\end{frame}

  \begin{itemize}
    \item alocar largura de banda fixa (não compartilhável) a um fluxo: uso ineficaz se o fluxo não usa a sua alocação
  \end{itemize}

  \begin{block}{Princípio 3}
    Ao proporcionar isolamento, é desejável usar os recursos da forma mais eficiente possível.
  \end{block}
\end{frame}

  \begin{columns}

      \textbf{Admissão de chamada}
      \begin{itemize}

\section{Mecanismos de escalonamento e regulação}

\begin{frame}{Escalonamento: como se escolhe o próximo pacote}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{FIFO (First In First Out)}
      \begin{itemize}
        \item enviar na ordem de chegada à fila
        \item política de descarte: \emph{drop-tail}, prioridade, aleatória
      \end{itemize}
      \textbf{Escalonamento prioritário}
      \begin{itemize}
        \item transmite da fila com maior prioridade
        \item múltiplas classes com prioridades distintas
        \item a classe pode depender da marcação ou origem/destino IP, portas\ldots
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Varredura cíclica (round robin)}
      \begin{itemize}
        \item varre as filas das classes, atendendo uma de cada classe (se disponível)
      \end{itemize}
      \textbf{WFQ --- Weighted Fair Queueing}
      \begin{itemize}
        \item varredura cíclica generalizada
        \item cada classe recebe uma quantidade \textbf{ponderada} de serviço em cada ciclo
      \end{itemize}
      \begin{center}
        \begin{tikzpicture}[font=\scriptsize]
          \node[draw,fill=aneliseGris!15,rounded corners=1mm](cola){fila (espera)};
          \node[draw,fill=aneliseVerde!20,rounded corners=1mm,right=1.4cm of cola](link){enlace (servidor)};
          \node[draw,ellipse,fill=aneliseLaranja!25,above=0.6cm of link](s){escalonador};
          \draw[->,thick,aneliseLaranja] (s.south) -- (link.north);
          \node[above=0.5mm of cola,font=\tiny]{chegadas};
          \node[below=0.5mm of link,font=\tiny]{saídas};
        \end{tikzpicture}\\[1mm]
        {\scriptsize \textbf{Figura 5:} Escalonador seleciona o próximo pacote entre as filas de espera.}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Mecanismos de regulação (shaping)}
  \begin{block}{Objetivo}
    Limitar o tráfego para não exceder os parâmetros declarados.
  \end{block}
  \textbf{Três critérios comuns:}
  \begin{itemize}
    \item \textbf{taxa média} (longo prazo): quantos pacotes por unidade de tempo
      \begin{itemize}
        \item pergunta crucial: qual é o tamanho do intervalo? (100~pkt/s vs.\ 6000~pkt/min)
      \end{itemize}
    \item \textbf{taxa de pico}: ex.\ 6000~pkt/min média, 1500~pkt/min pico
    \item \textbf{tamanho do surto (máximo)}: n.º máximo de pacotes consecutivos
  \end{itemize}
  \begin{block}{Balde de fichas (token bucket)}
    \begin{itemize}
      \item limita tamanho de surto e taxa média
      \item o balde mantém fichas; geram-se fichas à taxa $r$ fichas/s (se o balde não está cheio)
      \item sobre o intervalo $t$: pacotes admitidos $\leq r\cdot t + b$
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Token bucket + WFQ $\Rightarrow$ atraso limitado}
  \begin{center}
    \begin{tikzpicture}[font=\scriptsize,
        c/.style={draw,rounded corners=2mm,fill=aneliseAzul!13,inner sep=1.2mm,text width=3cm,align=center},
      ]
      \node[c](tb){token bucket\\(taxa $r$, tamanho $b$)};
      \node[c,right=2.4cm of tb](wfq){WFQ\\(taxa por fluxo $R$)};
      \draw[->,thick,aneliseLaranja] (tb) -- node[midway,above=1mm,font=\tiny]{tráfego modelado} (wfq);
      \node[draw,fill=aneliseVerde!15,rounded corners=2mm,below=6mm of wfq](d){Atraso máx.\ $D_{\max}=b/R$};
      \draw[->,thick,aneliseVerde] (wfq.south) -- (d.north);
      \node[draw,fill=aneliseGris!12,rounded corners=2mm,below=6mm of tb](gar){Garantia de QoS};
      \draw[dashed,->] (tb.south) -- (gar.north);
    \end{tikzpicture}\\[1mm]
    {\scriptsize \textbf{Figura 6:} Token bucket + WFQ garante atraso máximo $D_{\max}=b/R$.}
  \end{center}
  \begin{itemize}
    \item o balde de fichas e WFQ combinam-se para fornecer \textbf{limite superior garantido no atraso}
    \item isto é a base dos serviços garantidos de Intserv
  \end{itemize}
  \note{Se $R \geq r$, a fila nunca excede $b$ bits; logo o atraso máximo de fila é $b/R$, independentemente da taxa média do fluxo. Este é o resultado-chave para o exercício resolvido.}
\end{frame}

\section{Diffserv e Intserv}

\begin{frame}{Serviços diferenciados da IETF (Diffserv)}
  \begin{itemize}
    \item querem classes de serviço ``qualitativas'': ``comporta-se como um fio''
    \item distinção de serviço relativa: \textbf{Platinum, Gold, Silver}
    \item escalabilidade: \textbf{funções simples no núcleo}, funções complexas na borda
  \end{itemize}
  \begin{center}
    \begin{tikzpicture}[font=\scriptsize,
        c/.style={draw,rounded corners=2mm,fill=aneliseAzul!10,inner sep=1mm,text width=3cm,align=center},
      ]
      \node[c](borda){Roteador de borda\\marcado por fluxo\\(perfil/fora de perfil)};
      \node[c,right=2.6cm of borda](nucleo1){Roteador de núcleo\\tráfego por classe\\buffering e escalonamento};
      \node[c,right=2.6cm of nucleo1](nucleo2){Roteador de núcleo\\(igual)};
      \draw[->,thick,aneliseLaranja] (borda) -- node[midway,above=1mm,font=\tiny]{preferência a pacotes no perfil} (nucleo1);
      \draw[->,thick,aneliseLaranja] (nucleo1) -- (nucleo2);
    \end{tikzpicture}\\[1mm]
    {\scriptsize \textbf{Figura 7:} Diferenciação e marcação na borda; tratamento por classe no núcleo.}
  \end{center}
  \begin{itemize}
    \item 6 bits de \textbf{DSCP} (Differentiated Services Code Point) no campo ToS/Traffic Class
    \item determinam o \textbf{PHB} (Per-Hop Behavior) no roteador
  \end{itemize}
\end{frame}

\begin{frame}{Classificação e condicionamento (Diffserv)}
  \begin{itemize}
    \item pacote marcado no campo \textbf{ToS} (IPv4) / \textbf{Traffic Class} (IPv6)
    \item 6 bits \textbf{DSCP} + 2 bits não usados (ECN)
    \item pode ser desejável limitar a taxa de injeção de tráfego de uma classe:
      \begin{itemize}
        \item o usuário declara \textbf{perfil de tráfego} (taxa, tamanho de surto)
        \item o tráfego mede-se; molda-se (shaping) se não está em conformidade
      \end{itemize}
  \end{itemize}
  \begin{block}{PHB (Per Hop Behavior)}
    \begin{itemize}
      \item PHB resulta em comportamento de encaminhamento diferente nos roteadores
      \item não especifica quais mecanismos usar
      \item exemplos: \textbf{EF} (Expedited Forwarding), \textbf{AF} (Assured Forwarding, 4 classes $\times$ 3 partições de descarte)
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Serviços integrados da IETF (Intserv)}
  \begin{itemize}
    \item arquitetura para garantir QoS em redes IP para \textbf{sessões de aplicação individuais}
    \item \textbf{reserva de recursos}: os roteadores mantêm estado (tipo VC) de recursos alocados
    \item \textbf{admitir/negar} novos requerimentos de chamada
  \end{itemize}
  \begin{block}{Admissão de chamada}
    A sessão que chega deve:
    \begin{enumerate}
      \item declarar o seu requerimento de QoS: \textbf{Rspec}
      \item caracterizar o tráfego: \textbf{Tspec}
      \item executar o protocolo de sinalização: \textbf{RSVP}
    \end{enumerate}
  \end{block}
  \begin{exampleblock}{Modelos de serviço}
    \textbf{Serviço garantido} (RFC 2212): chegada regulada por leaky bucket; atraso limitado matematicamente.
    \textbf{Carga controlada} (RFC 2211): ``QoS próxima à de uma rede não carregada''.
  \end{exampleblock}
\end{frame}

\begin{frame}{RSVP --- Resource Reservation Protocol}
  \begin{block}{Motivação}
    Encaminhamento sem conexão + melhor esforço = sem sinalização no projeto inicial do IP.
    Novo requerimento: reservar recursos fim-a-fim para QoS em multimídia.
  \end{block}
  \begin{itemize}
    \item \textbf{RSVP} [RFC 2205]: ``permitir que os usuários comuniquem requerimentos à rede de modo robusto e eficaz''
    \item antigo protocolo de sinalização: \textbf{ST-II} [RFC 1819]
  \end{itemize}
  \begin{block}{RSVP não\ldots}
    \begin{itemize}
      \item não especifica como reservar recursos (é um mecanismo de comunicação de necessidades)
      \item não determina as rotas (tarefa do roteamento); sinalização desacoplada do roteamento
      \item não interage com o encaminhamento de pacotes; separação dos planos de controle e dados
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{RSVP: visão geral da operação}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item remetentes e receptores unem-se a um grupo multicast (fora do RSVP)
        \item \textbf{sinalização do remetente à rede}:
          \begin{itemize}
            \item mensagem \texttt{PATH}: faz conhecida a presença do remetente
            \item \texttt{PATH teardown}: elimina o estado do caminho
          \end{itemize}
        \item \textbf{sinalização do receptor à rede}:
          \begin{itemize}
            \item mensagem \texttt{RESV}: reserva recursos do(s) remetente(s)
            \item \texttt{RESV teardown}: elimina reservas
          \end{itemize}
        \item \textbf{sinalização da rede aos sistemas finais}:
          \begin{itemize}
            \item erro de caminho, erro de reserva
          \end{itemize}
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Objetivos de projeto}:
      \begin{enumerate}
        \item acomodar receptores heterogêneos
        \item acomodar aplicações com distintos requerimentos
        \item multicast como serviço de primeira classe
        \item aproveitar roteamento multicast/unicast
        \item overhead de protocolo linear no n.º de receptores
        \item projeto modular
      \end{enumerate}
    \end{column}
  \end{columns}
  \note{Enfatizar que o RSVP é orientado ao receptor (receiver-initiated): cada receptor envia RESV e o estado é mantido por fluxo, o que limita a escalabilidade em redes grandes.}
\end{frame}

\section{QoS em redes sem fio}

\begin{frame}{QoS em redes sem fio}
  \begin{itemize}
    \item ideia: prover QoS em redes móveis e ad hoc com o \textbf{mínimo overhead possível}
    \item podem-se usar os modelos de QoS das redes cabeadas?
    \item duas frentes: \textbf{QoS routing} e \textbf{QoS em Wi-Fi}
  \end{itemize}
  \begin{block}{QoS Routing}
    \begin{itemize}
      \item o roteamento é componente essencial: informa à fonte a banda e a QoS disponível rumo ao destino
      \item múltiplas rotas podem oferecer-se com diferentes requerimentos de QoS
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{QOLSR --- QoS para OLSR}
  \begin{itemize}
    \item OLSR é proativo, padronizado pela IETF (RFC 3626)
    \item QoS para OLSR proposto em 2003
    \item ideia principal: usar \textbf{métricas de QoS na seleção dos MPRs}
    \item as medidas de QoS obtêm-se com as próprias mensagens \texttt{HELLO} --- \textbf{sem novas mensagens de controle}
  \end{itemize}
  \begin{block}{Heurísticas QOLSR}
    \begin{itemize}
      \item QOLSR1: seleciona o vizinho de 1 salto com menor atraso como MPR
      \item QOLSR2: antecipa além disso o 2.º salto (mínimo atraso 2-hop)
      \item QOLSR3: escolhe MPR que alcance o maior número de nós descobertos
      \item não há garantia de caminho ótimo em relação à métrica
    \end{itemize}
  \end{block}
  \begin{exampleblock}{Referência}
    A.\ Munaretto, M.\ Fonseca, ``Routing and quality of service support for mobile ad hoc networks'', \emph{Computer Networks} 51:3142--3156, 2007.
  \end{exampleblock}
\end{frame}

\begin{frame}{QoS em Wi-Fi: anomalia de desempenho}
  \begin{block}{Performance Anomaly (802.11b)}
    \begin{itemize}
      \item numa rede 802.11b, um único nó lento (baixa SNR) reduz o throughput de todos os demais
      \item causa: CSMA/CA dá \textbf{igual oportunidade} de acesso independentemente da taxa
      \item o tempo do canal consome-se com o nó lento
    \end{itemize}
  \end{block}
  \begin{exampleblock}{Solução: Fair Time Sharing (FTS)}
    \begin{itemize}
      \item atribuir \textbf{tempo de canal} proporcional à taxa do nó (não slots iguais)
      \item A.\ Munaretto, M.\ Fonseca, ``Self-adapting algorithm to fair time sharing in wireless access networks'', \emph{Computers \& Electrical Engineering} 34:406--415, 2008
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{802.11e --- QoS no MAC}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{EDCF / EDCA}
      \begin{itemize}
        \item 4 \textbf{Access Categories} (AC): Voice, Video, Best-Effort, Background
        \item cada AC tem o seu \textbf{AIFS} e o seu \textbf{Backoff Counter}
        \item colisão virtual: a maior prioridade ganha, as outras incrementam CW
        \item \textbf{TXOP}: intervalo de transmissão sem contenção

\section{Conclusões}

\begin{frame}{Conclusões}
  \begin{alertblock}{Ideia central}
    Prover garantias de QoS em redes sem fio é uma tarefa \textbf{complexa}!
  \end{alertblock}
  \begin{block}{Uma solução completa de QoS requer}
    \begin{enumerate}
      \item um modelo apropriado de QoS
      \item um protocolo de sinalização de QoS
      \item um protocolo de roteamento com QoS
      \item um protocolo de acesso ao meio com QoS
      \item mecanismos adicionais: CAC, gestor de políticas, filas para controle de congestionamento\ldots
    \end{enumerate}
  \end{block}
  \begin{exampleblock}{Atualização 2026}
    As lições de Intserv/Diffserv/RSVP vivem em 5G \emph{network slicing} e no Wi-Fi 6 (OFDMA + QoS).
  \end{exampleblock}
\end{frame}

\begin{frame}{Exercício resolvido: token bucket + WFQ --- atraso de fila}
  \textbf{Dados}: \emph{token bucket} com taxa média $r=1$~Mbps e balde $b=1$~Mb; WFQ com taxa garantida $R=1{,}5$~Mbps ao fluxo; pacotes de $1500$ bytes ($=12\,000$ bits).
  \begin{block}{Parte A --- Atraso máximo de fila}
    Como $R\geq r$, a fila nunca excede $b$ bits: $D_{\max}=b/R=10^6/(1{,}5\times 10^6)=2/3\approx \mathbf{667}$~\textbf{ms}
  \end{block}
  \begin{block}{Parte B --- Capacidade do surto}
    \begin{itemize}
      \item pacotes no surto máximo: $N=b/(1500\times 8)=10^6/12\,000\approx 83$ pacotes
      \item tempo de transmissão de 1 pacote: $t_p=12\,000/(1{,}5\times 10^6)=8$~ms
      \item verificação: $83\times 8=664$~ms $\approx D_{\max}$ \checkmark
    \end{itemize}
  \end{block}
  \begin{alertblock}{Interpretação}
    O atraso de fila é \textbf{limitado pelo surto} $b$, não pela taxa média: é a base das garantias do \emph{serviço garantido} do Intserv (RFC 2212).
  \end{alertblock}
\end{frame}

\begin{frame}{Resumo da aula}
  \begin{itemize}
    \item \textbf{Princípios de QoS}: marcação, \emph{shaping}, compartilhamento eficiente e admissão de chamadas;
    \item \textbf{Intserv + RSVP}: reserva fim-a-fim por fluxo (PATH/RESV, Tspec/Rspec); \textbf{Diffserv}: classes com DSCP/PHB, funções complexas na borda;
    \item \textbf{Token bucket + WFQ}: garantia de atraso máximo $D_{\max}=b/R$, base do serviço garantido;
    \item \textbf{Redes sem fio}: QOLSR usa métricas de QoS na seleção de MPRs; FTS aloca tempo de canal proporcional à taxa; 802.11e/EDCA prioriza classes (ACs);
    \item \textbf{Visão 2026}: os conceitos persistem em \emph{network slicing} 5G e na diferenciação do Wi-Fi 6 (OFDMA + QoS).
  \end{itemize}
\end{frame}
