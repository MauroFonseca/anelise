\section{O que é a Internet?}

% ---------------------------------------------------------------------------
\begin{frame}{O que é a Internet: vista \emph{nuts and bolts}}
  \textbf{Bilhões} de dispositivos de computação conectados:
  \begin{itemize}
    \item \textbf{\emph{Hosts} = sistemas finais} (\emph{end systems}) que executam aplicações de rede
    \item \textbf{Enlaces de comunicação}: fibra, cobre, rádio, satélite
      \begin{itemize}
        \item taxa de transmissão = \alert{largura de banda} (bps)
      \end{itemize}
    \item \textbf{Comutadores de pacotes}: \textbf{roteadores} e \textbf{switches} que reenviam \emph{chunks} de dados (pacotes)
  \end{itemize}

% ---------------------------------------------------------------------------
\begin{frame}{O que é a Internet: vista de \emph{serviço}}
  \begin{block}{Infraestrutura que fornece serviços às aplicações}
    Web, VoIP, e-mail, jogos, comércio eletrônico, redes sociais, \emph{streaming}, \ldots
  \end{block}
  \begin{itemize}
    \item Fornece \textbf{interface de programação} às aplicações
      \begin{itemize}
        \item \emph{hooks} que permitem que programas de aplicação ``se conectem'' à Internet
        \item análogo ao \textbf{serviço postal}: oferece opções de serviço
      \end{itemize}
  \end{itemize}
  \begin{exampleblock}{Vista atualizada (2026)}
    A Internet também é uma \textbf{plataforma de computação}: \emph{computing in the network} (middleboxes, CDNs, \emph{edge computing}, segmentação de rede 5G).
  \end{exampleblock}
  \note{Destaque a analogia com o serviço postal; retome a ideia de ``se conectar'' aos \emph{hooks} das aplicações.}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Protocolos e padrões}
  \begin{itemize}
    \item \textbf{Protocolos} controlam o envio e recepção de mensagens:
      \begin{itemize}
        \item e.g., TCP, UDP, IP, HTTP/3 (QUIC), TLS, 802.11 (Wi-Fi), 5G NR
      \end{itemize}
    \item \textbf{Padrões da Internet}:
      \begin{itemize}
        \item \textbf{RFC} --- \emph{Request for Comments} (documentos que definem os protocolos)
        \item \textbf{IETF} --- \emph{Internet Engineering Task Force} (organização que publica RFCs)
        \item \textbf{IANA} --- números de protocolo, portas, etc.
      \end{itemize}
  \end{itemize}

\section{Camada de protocolos}

% ---------------------------------------------------------------------------
\begin{frame}{Por que \emph{layering} (camadas)?}
  As redes são complexas e têm muitas ``peças'' (hosts, roteadores, enlaces, aplicações, protocolos\ldots).
  \begin{block}{Vantagens}
    \begin{itemize}
      \item \textbf{Estrutura explícita} identifica e relaciona as partes de um sistema complexo
      \item \textbf{Modularização} facilita manutenção e atualização
        \begin{itemize}
          \item mudar a implementação de uma camada é \emph{transparente} para o resto do sistema
        \end{itemize}
    \end{itemize}
  \end{block}
  \begin{alertblock}{Discussão}
    \emph{``Layering considered harmful?''} A estratificação rígida pode limitar o desempenho (e.g., QUIC funde funções de transporte $+$ segurança); por isso a pesquisa moderna explora o \emph{cross-layer design}.
  \end{alertblock}
\end{frame}

\begin{frame}{Plano de dados vs.\ plano de controle (SDN)}
  \begin{block}{Duas funções da camada de rede}
    \begin{itemize}
      \item \textbf{Encaminhamento (\emph{forwarding})}: mover pacotes da entrada à saída apropriada do roteador \quad (função local --- \alert{plano de dados})
      \item \textbf{Roteamento (\emph{routing})}: determinar a rota origem$\rightarrow$destino \quad (função global --- \alert{plano de controle})
    \end{itemize}
  \end{block}
  \begin{columns}
    \begin{column}{0.44\textwidth}
      \textbf{Abordagem tradicional:}
      \begin{itemize}
        \item plano de controle implementado \emph{no próprio roteador} (protocolos distribuídos: RIP, OSPF, BGP)
      \end{itemize}
      \textbf{Abordagem SDN} (\emph{Software-Defined Networking}):
      \begin{itemize}
        \item plano de controle \emph{centralizado logicamente} em servidores remotos (controlador SDN)
        \item o roteador ``burro'' só encaminha segundo a tabela de fluxos (OpenFlow)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}

    \end{column}
  \end{columns}
\end{frame}

\section{Endereçamento IP}

\begin{frame}{Endereçamento IP: introdução}
  \begin{itemize}
    \item Endereço IP: \textbf{identificador de 32 bits} para a interface de host/roteador
    \item \textbf{Interface}: conexão entre host/roteador e enlace físico

\begin{frame}{Subredes (subnets)}
  \begin{itemize}
    \item Endereço IP $=$ \textbf{parte de subrede} (bits altos) + \textbf{parte de host} (bits baixos)
    \item \textbf{Subrede}: interfaces de dispositivo com a mesma parte de subrede

\section{Roteamento}

% ---------------------------------------------------------------------------
\begin{frame}{Protocolos de roteamento: objetivo}
  \begin{block}{Objetivo}
    Determinar caminhos ``bons'' (\emph{routes}) da origem ao destino através da rede de roteadores.
  \end{block}
  \begin{itemize}
    \item \textbf{Caminho}: sequência de roteadores que os pacotes atravessarão
    \item ``Bom'': menor ``custo'', ``mais rápido'', ``menos congestionado''
    \item \alert{``Roteamento é um dos top-10 desafios de rede!''}
  \end{itemize}
  \begin{exampleblock}{Escalabilidade --- o problema}
    \begin{itemize}
      \item com \emph{milhões} de destinos não se podem guardar todas as rotas nas tabelas
      \item a troca de tabelas de roteamento saturaria os enlaces
      \item \textbf{autonomia administrativa}: cada rede quer controlar o próprio roteamento
    \end{itemize}
  \end{exampleblock}
\end{frame}

\section{Camada de enlace}

% ---------------------------------------------------------------------------
\begin{frame}{Camada de enlace: introdução}
  \begin{itemize}
    \item \textbf{Nós}: hosts e roteadores
    \item \textbf{Enlaces}: canais de comunicação entre nós \emph{adjacentes}
      \begin{itemize}
        \item cabeado, sem fio, LANs
      \end{itemize}
    \item Unidade da camada 2: \textbf{frame} (quadro), que encapsula um datagrama
    \item Responsabilidade: transferir o datagrama de um nó ao nó \emph{fisicamente adjacente}
  \end{itemize}
  \begin{block}{Analogia do transporte}
    Viagem Princeton $\rightarrow$ Lausana: limusina (Princeton$\rightarrow$JFK), avião (JFK$\rightarrow$Genève), trem (Genève$\rightarrow$Lausanne).
    \begin{itemize}
      \item turista $=$ datagrama \quad transporte $=$ enlace \quad modo $=$ protocolo de enlace \quad agente de viagens $=$ algoritmo de roteamento
    \end{itemize}
  \end{block}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Serviços da camada de enlace}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Enquadramento e acesso ao meio:}
      \begin{itemize}
        \item encapsular datagrama em frame (header+trailer)
        \item acesso ao canal se o meio é compartilhado
        \item endereços \textbf{MAC} no header para identificar origem/destino (diferente do IP!)
      \end{itemize}
      \textbf{Entrega confiável entre nós adjacentes:}
      \begin{itemize}
        \item rara em fibra/par trançado (baixa BER)
        \item essencial em enlaces sem fio (altas taxas de erro)
        \item \alert{Q: por que confiabilidade a nível de enlace \emph{e} fim-a-fim?}
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Controle de fluxo:}
      \begin{itemize}
        \item sincronização entre nó transmissor e receptor adjacentes
      \end{itemize}
      \textbf{Detecção e correção de erros:}
      \begin{itemize}
        \item \emph{error detection}: receptor detecta erros $\rightarrow$ retransmissão ou descarte
        \item \emph{error correction}: receptor corrige bit(s) sem retransmissão (FEC)
      \end{itemize}
      \textbf{Half/full-duplex:}
      \begin{itemize}
        \item half-duplex: ambos transmitem, mas não simultaneamente
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Endereços MAC e ARP}
  \begin{itemize}

\begin{frame}{Ethernet CSMA/CD (legado) --- algoritmo}
  \begin{enumerate}
    \item A NIC recebe o datagrama da camada de rede e cria o frame
    \item Se a NIC \textbf{sente o canal ocioso} (\emph{carrier sense}), inicia a transmissão; se ocupado, espera
    \item Se transmite o frame \textbf{completo} sem detectar outra transmissão $\rightarrow$ \alert{feito!}
    \item Se detecta outra transmissão \textbf{durante a transmissão} $\rightarrow$ aborta e envia \emph{jam signal} (48 bits)
    \item \textbf{Backoff exponencial}: após a $m$-ésima colisão, escolhe $K\in\{0,1,\ldots,2^m-1\}$, espera $K\cdot 512$ bit-times e volta ao passo 2
  \end{enumerate}
  \begin{alertblock}{Estado atual (2026)}
    Ethernet moderno é \textbf{full-duplex ponto-a-ponto} (switches): CSMA/CD ficou como material histórico; o legado vive nas LANs sem fio em forma de \alert{CSMA/CA} (Aula~4).
  \end{alertblock}
  \note{Armadilha comum: confundir CSMA/CD (cabeado, com detecção) com CSMA/CA (sem fio, com prevenção); a aula 4 retoma o tema sem fio.}
\end{frame}

\section{ICMP e Traceroute}

% ---------------------------------------------------------------------------
\begin{frame}{ICMP --- Internet Control Message Protocol}
  \begin{itemize}
    \item Usado por hosts e roteadores para comunicar informações de nível de rede
      \begin{itemize}
        \item \emph{error reporting}: host/red/porta/protocolo inalcançáveis
        \item echo request/reply (base do \texttt{ping})
      \end{itemize}
    \item Mensagens ICMP viajam \textbf{dentro de datagramas IP}
    \item Mensagem: \texttt{type} + \texttt{code} + primeiros 8 bytes do datagrama que causou o erro
  \end{itemize}
  \vspace{0.1cm}
  \begin{center}
    \scriptsize
    \begin{tabular}{@{}lll@{}}
      \textbf{Tipo} & \textbf{Código} & Descrição\\
      0 & 0 & echo reply (ping)\\
      3 & 0/1/2/3 & destino rede/host/protocolo/porta inalcançável\\
      3 & 6/7 & rede/host desconhecido\\
      4 & 0 & source quench (congestão; \alert{obsoleto})\\
      8 & 0 & echo request (ping)\\
      9/10 & 0 & route advertisement / router discovery\\
      11 & 0 & TTL expirado (\alert{traceroute!})\\
      12 & 0 & cabeçalho IP malformado\\
    \end{tabular}
  \end{center}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Traceroute e ICMP: como funciona}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Origem envia \textbf{sequências de segmentos UDP} em direção ao destino:
          \begin{itemize}
            \item 1ª seq.\ com \texttt{TTL=1}, 2ª com \texttt{TTL=2}, \ldots
            \item porta pouco provável
          \end{itemize}
        \item Quando o datagrama da $n$-ésima seq.\ chega ao $n$-ésimo roteador:
          \begin{itemize}
            \item o roteador descarta o datagrama e devolve \textbf{ICMP tipo 11 código 0}
            \item inclui nome e IP do roteador
          \end{itemize}
        \item Origem registra os \textbf{RTTs} (3 probes por salto)
      \end{itemize}
      \textbf{Critério de parada:}
      \begin{itemize}
        \item o segmento UDP chega ao destino; o destino devolve \textbf{ICMP tipo 3 código 3} (porta inalcançável)
      \end{itemize}
    \end{column}

\begin{frame}{Exercício resolvido: latência de um enlace}
  \textbf{Problema:} Um arquivo de \textbf{1.500 bytes} é transmitido por um enlace de fibra óptica com taxa $R = 100$~Mbps e comprimento $d = 100$~km ($s = 2\times10^8$~m/s). Calcule o atraso de transmissão, o de propagação e o atraso total.
  \pause
  \begin{block}{Solução passo a passo}
    \begin{align*}
      d_{trans} &= \frac{L}{R} = \frac{1.500 \times 8}{100 \times 10^6}
      = \frac{12.000}{10^8} = 0{,}12~\text{ms},\\[2mm]
      d_{prop} &= \frac{d}{s} = \frac{100 \times 10^3}{2 \times 10^8}
      = 0{,}5~\text{ms},\\[2mm]
      d_{total} &= d_{trans} + d_{prop} = 0{,}12 + 0{,}5
      = \alert{0{,}62~\text{ms}}.
    \end{align*}
  \end{block}
  \pause
  \textbf{Conclusão:} a propagação domina (0,5~ms) --- comportamento típico em enlaces longos de alta taxa.
\end{frame}

\begin{frame}{Resumo da aula}
  \begin{itemize}
    \item A Internet é uma \textbf{rede de redes}: hosts, enlaces e comutadores de pacotes (roteadores/switches).
    \item Protocolos (\textbf{TCP, IP, HTTP, DNS}) e padrões (\textbf{RFC, IETF, IANA}) definem a comunicação.
    \item O modelo TCP/IP de \textbf{5 camadas} organiza a comunicação; o modelo OSI tem 7 camadas.
    \item Cada camada \textbf{encapsula} os dados com seus cabeçalhos.
    \item \textbf{Endereçamento}: IP (32/128 bits) para a rede e MAC (48 bits) para o enlace; o ARP resolve IP$\rightarrow$MAC.
    \item O \textbf{roteamento} intra-AS e inter-AS (BGP) escala a Internet em sistemas autônomos.
    \item \textbf{ICMP} e \textbf{traceroute} permitem diagnosticar a rede; o \textbf{Wireshark} revela o encapsulamento em todas as camadas.
  \end{itemize}
\end{frame}
