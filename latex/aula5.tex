\section{IEEE 802.11 (Wi-Fi)}

\begin{frame}{LAN sem fio IEEE 802.11: evolução}
  \begin{center}
    \scriptsize
    \begin{tabular}{@{}llll@{}}
      \toprule
      \textbf{Padrón} & \textbf{Banda} & \textbf{Tasa} & \textbf{Notas}\\
      \midrule
      802.11b & 2,4~GHz & 11~Mbps & DSSS, chip de 11\\
      802.11a & 5--6~GHz & 54~Mbps & OFDM (52 subport.)\\
      802.11g & 2,4--5~GHz & 54~Mbps & OFDM\\
      802.11n (Wi-Fi 4) & 2,4/5~GHz & 600~Mbps & MIMO 4$\times$4, OFDM 108\\
      802.11ac (Wi-Fi 5) & 5~GHz & 6,9~Gbps & MIMO 8$\times$8, QAM-256\\
      802.11ax (Wi-Fi 6) & 2,4/5/6~GHz & 9,6~Gbps & OFDMA, QAM-1024, MU-MIMO\\
      802.11be (Wi-Fi 7) & 2,4/5/6~GHz & $\sim$46~Gbps & 320~MHz, QAM-4096, mMLS\\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item todos usan \textbf{CSMA/CA} para acceso múltiple
    \item todos têm versión de estação base e ad hoc
  \end{itemize}
\end{frame}

\begin{frame}{802.11: canais e asociación}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{802.11b}: espectro 2,4--2,485~GHz dividido em 11 canais
      \begin{itemize}
        \item o admin do AP escolle a frequência do AP
        \item interferência posible: o canal pode ser o mesmo que o do AP veciño!
      \end{itemize}
      \textbf{Asociación do hospedador}
      \begin{enumerate}
        \item varre os canais, escoitando \textbf{beacons} (SSID e MAC)
        \item selecciona un AP para asociarse
        \item pode facer autenticación (WPA3)
        \item normalmente executa DHCP para obter IP na subrede do AP
      \end{enumerate}
    \end{column}

\begin{frame}{Wi-Fi 6 (802.11ax) e Wi-Fi 6E}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Wi-Fi 6 (802.11ax, 2019)}
      \begin{itemize}
        \item \textbf{OFDMA}: subportadoras asignadas a múltiples usuários no mesmo slot
        \item \textbf{MU-MIMO} uplink/downlink
        \item QAM-1024
        \item objetivo: eficiencia em ambientes densos (estadios, aeroportos)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Wi-Fi 6E (2021)}
      \begin{itemize}
        \item engade a \textbf{banda de 6~GHz} (5925--7125~MHz)
        \item mais largura de banda dispoñible, menos interferência
        \item requer \textbf{WPA3} obrigatorio
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{block}{Wi-Fi 7 (802.11be, 2024)}
    \begin{itemize}
      \item canais de \textbf{320~MHz}, QAM-4096, \emph{multi-link operation} (MLO)
      \item throughput teórico ata \textbf{46~Gbps} (em ambientes ideais)
      \item dirixido a 8K streaming, AR/VR, TSN/industria
    \end{itemize}
  \end{block}
\end{frame}

\section{LPWAN: LoRa e Sigfox}

\begin{frame}{LPWAN para IoT: LoRa vs.\ Sigfox}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{LoRa / LoRaWAN}
      \begin{itemize}
        \item modulación \textbf{Chirp Spread Spectrum (CSS)}
        \item banda ISM sub-GHz (868~MHz EU, 915~MHz US/BR)
        \item tasas 0,3--50~kbps; alcance 2--15~km
        \item topoloxía \emph{star-of-stars}: sensores $\rightarrow$ gateways $\rightarrow$ servidor
        \item ALOHA puro + canais com \emph{duty cycle}
        \item autenticación AES-128 (join E2E)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Sigfox}
      \begin{itemize}
        \item banda ultra-estreita (UNB, 100~Hz)
        \item uplink 100 bps; 140 mensagens/día
        \item alcance ata 50~km (rural)
        \item rede propietaria (operador rexional)
      \end{itemize}
      \begin{block}{Comparación práctica}
        LoRaWAN: mais flexible, redes privadas, bidireccional.
        Sigfox: ultra-sinxe-lo, moi baixo consumo, unidireccional.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{NB-IoT e eMTC (celular LPWA)}
  \begin{itemize}
    \item estándares 3GPP para IoT masivo dentro de LTE/5G:
      \begin{itemize}
        \item \textbf{NB-IoT} (Narrowband IoT): 200~kHz, ata 250~kbps, moi boa penetración indoor
        \item \textbf{LTE-M / eMTC}: 1~MHz, ata 1~Mbps, soporta voice e mobility
      \end{itemize}
    \item comparados com LoRa: maior confiabilidade, QoS garantida, pero maior consumo e custo de módulo
  \end{itemize}

\section{5G e redes móveis}

\begin{frame}{5G New Radio (NR)}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Tres casos de uso}
      \begin{itemize}
        \item \textbf{eMBB} (Enhanced Mobile Broadband): $>$ 1~Gbps, 8K/VR
        \item \textbf{uRLLC} (Ultra-Reliable Low Latency): 1~ms, industria, condución autónoma
        \item \textbf{mMTC} (Massive Machine-Type): 1M dispositivos/km², IoT masivo
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Tecnoloxías clave}
      \begin{itemize}
        \item \emph{beamforming} masivo (Massive MIMO)
        \item \emph{network slicing} (redes virtuais por servizo)
        \item \emph{edge computing} (MEC)
        \item ondas milimétricas (mmWave 26--28~GHz, 60~GHz)
      \end{itemize}
    \end{column}
  \end{columns}
  \begin{exampleblock}{Evolución}
    5G-Advanced (3GPP Rel.\ 18, 2024): IA/ML na rede, RedCap. O 6G xa está em investigación (3GPP Rel.\ 21, cara 2030).
  \end{exampleblock}
\end{frame}

    \begin{column}{0.5\textwidth}
      \textbf{DTN (Delay/Disruption Tolerant Networks)}
      \begin{itemize}
        \item tolerancia a atrasos e interrupcións (espaço, móbil rural)
        \item \emph{store-and-forward} co protocolo \textbf{Bundle Protocol (RFC 5050)}
      \end{itemize}
      \textbf{Flying Networks (drones)}
      \begin{itemize}
        \item FANET: enxames de UAV, roteamento 3D, mobilidade rápida
      \end{itemize}
      \textbf{Vehicular Networks}
      \begin{itemize}
        \item C-V2X (celular) vs.\ 802.11p/ITS-G5
        \item comunicação V2V, V2I, V2P
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Satélite e NTN}

\begin{frame}{Satélite: redes não terrestres (NTN)}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Satélite tradicional}
      \begin{itemize}
        \item GEO (35\,786~km): cobertura ampla, alta latencia (125~ms)
        \item MEO/LEO: menor latencia
      \end{itemize}
      \textbf{LEO masivo (2026)}
      \begin{itemize}
        \item \textbf{Starlink} (SpaceX): $>$ 7000 satélites, constelación em LEO $\sim$550~km
        \item \textbf{OneWeb, Kuiper, Telesat}
        \item latencia 20--40~ms; throughput 100--300~Mbps por terminal
        \item láser ISL (inter-satellite links); integración 5G NTN (3GPP Rel.\ 17+)
      \end{itemize}

\section{SDN, radio cognitivo e investigación}

\begin{frame}{Radio cognitivo e SDN}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Radio cognitivo (IEEE 802.22 / TVWS)}
      \begin{itemize}
        \item acceso oportunista ao espectro (escanear, detectar, usar)
        \item \emph{cognitive radio networks}: espectro dinámico
      \end{itemize}
      \textbf{SDR (Software Defined Radio)}
      \begin{itemize}
        \item procesado do sinal por software; axilidade espectral
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{SDN (Software Defined Networking)}
      \begin{itemize}
        \item plano de control centralizado
        \item aplicado a WMN, VANET, IoT
        \item vantaxe: programabilidade da rede
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Proxectos de investigación em redes sem fio}
  \textbf{Áreas principais}: optimización de recursos e protocolos de roteamento.
  \begin{enumerate}
    \item Redes complexas $\times$ datos dinámicos $\times$ Big Data $\times$ IA
    \item IoT e Smart Cities: LoRaONE com CSMA
    \item Redes ad hoc de drones $\times$ gestão de mobilidade $\times$ optimización de recursos
    \item Intelixencia colectiva $\times$ roteamento $\times$ redes sem fio oportunistas $\times$ escenarios urbanos
    \item Redes veiculares $\times$ mobilidade urbana $\times$ roteamento
    \item Optimización de recursos $\times$ radios cognitivos
    \item Internet usando a rede de transporte público de Curitiba
  \end{enumerate}
  \begin{exampleblock}{Renovación}
    Incorporar: \emph{mesh Wi-Fi 6/7 doméstico}, \emph{network slicing 5G}, \emph{TPU/GPU sensing}, \emph{dataset abertos de Curitiba} (UBRS).
  \end{exampleblock}
\end{frame}
