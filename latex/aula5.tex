\begin{frame}{Wi-Fi 6 (802.11ax) e Wi-Fi 6E}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Wi-Fi 6 (802.11ax, 2019)}
      \begin{itemize}
        \item \textbf{OFDMA}: subportadoras alocadas a múltiplos usuários no mesmo slot
        \item \textbf{MU-MIMO} uplink/downlink
        \item QAM-1024
        \item objetivo: eficiência em ambientes densos (estádios, aeroportos)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Wi-Fi 6E (2021)}
      \begin{itemize}
        \item adiciona a \textbf{banda de 6~GHz} (5925--7125~MHz)
        \item mais largura de banda disponível, menos interferência
        \item requer \textbf{WPA3} obrigatório
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \pause
  \begin{block}{Wi-Fi 7 (802.11be, 2024)}
    \begin{itemize}
      \item canais de \textbf{320~MHz}, QAM-4096, \emph{multi-link operation} (MLO)
      \item throughput teórico até \textbf{46~Gbps} (em ambientes ideais)
      \item dirigido a streaming 8K, AR/VR, TSN/indústria
    \end{itemize}
  \end{block}
  \note{Enfatizar que o MLO do Wi-Fi 7 usa simultaneamente as bandas de 2,4/5/6 GHz, reduzindo latência e aumentando confiabilidade; o Wi-Fi 6E foi o primeiro a habilitar a banda de 6 GHz.}
\end{frame}

    \item trade-off: RU-26 $\Rightarrow$ mais usuários simultâneos, menos taxa; RU-242 $\Rightarrow$ mais taxa para 1 usuário.
  \end{itemize}

\begin{frame}{Wi-Fi 7 (802.11be): Multi-Link Operation (MLO)}
  \small
  \begin{columns}
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item a STA opera \textbf{simultaneamente} em \textbf{2,4/5/6~GHz}, com um \emph{link lógico} sobre vários enlaces físicos;
        \item \textbf{Throughput}: agrega capacidade de várias bandas no mesmo instante;
        \item \textbf{Latência}: o tráfego crítico usa o enlace mais rápido --- atrasos menores que no ax/ac;
        \item \textbf{Confiabilidade}: se um enlace degrada, \emph{failover} automático para outro, sem reconexão;
        \item modos de operação: \emph{STR} (transmitir/receber simultâneo) e \emph{NSTR}.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{center}

    \end{column}
  \end{columns}
  \begin{alertblock}{Na prática}
    MLO reduz latência (jogos, AR/VR) e aumenta a vazão; o Wi-Fi 6E habilitou a banda de 6~GHz.
  \end{alertblock}
\end{frame}

\begin{frame}{Wi-Fi 8 (802.11bn): o que esperar}
  \begin{itemize}
    \item \textbf{previsão de conclusão}: $\sim$2028 (IEEE 802.11bn, ainda em estudo);
    \item \textbf{espectro}: amplia o uso de canais de \textbf{320~MHz} e do MLO multi-enlace (até 4 enlaces coordenados);
    \item \textbf{modulação}: mantém o \textbf{4096-QAM} introduzido no Wi-Fi 7 e melhora a codificação e a eficiência espectral;
    \item \textbf{latência}: metas de \textbf{baixa latência determinística} (XR, jogos em nuvem, automação industrial);
    \item \textbf{coordenação}: aprimoramentos de OFDMA e \emph{multi-AP coordination} para ambientes densos.
  \end{itemize}
  \begin{exampleblock}{Cuidado com a leitura}
    O Wi-Fi 7 \emph{já} tem QAM-4096 e canais de 320~MHz; a novidade do Wi-Fi 8 está na \textbf{latência} e na \textbf{coordenação}, não apenas na taxa bruta.
  \end{exampleblock}
\end{frame}

\begin{frame}{Bluetooth Low Energy (BLE) --- LE na prática}
  \small
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \textbf{Arquitetura}
      \begin{itemize}
        \item 40 canais de 2~MHz em 2,4~GHz: 3 de \textbf{advertising} (37/38/39) + 37 de dados com \textbf{salto adaptativo};
        \item papéis \emph{central}/\emph{peripheral} e \emph{advertiser}/\emph{scanner}; GAP (descoberta) e GATT (serviços);
        \item conexão: \emph{connection events} a cada 7,5~ms--4~s, com \emph{slave latency}.
      \end{itemize}
      \textbf{Broadcasting (advertising)}
      \begin{itemize}
        \item pacotes pequenos, periódicos, \textbf{sem conexão} (beacons de proximidade, \emph{trackers}).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Consumo vs.\ BR/EDR}
      \begin{itemize}
        \item LE: pico $\sim$10--15~mA, média de \textbf{µA}; anos com pilha de botão;
        \item BR/EDR: dezenas de mA ativos --- pior para sensores.
      \end{itemize}
      \textbf{Taxas LE}
      \begin{itemize}
        \item LE 1M (1~Mbps), LE 2M (2~Mbps) e \emph{Coded PHY} (500/125~kbps para alcance);
        \item Bluetooth 5.x: \emph{LE Audio}, mesh e \emph{long range}.
      \end{itemize}
      \begin{alertblock}{Quando usar}
        Leitura discreta e sensores de curto alcance --- não para streaming pesado.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{UWB --- IEEE 802.15.4z (posicionamento em cm)}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \textbf{Física}
      \begin{itemize}
        \item pulsos de curta duração ($\sim$2~ns) e banda larga ($\ge$500~MHz), faixa de 6--9~GHz;
        \item \textbf{tempo de voo} (ToF) com \emph{double-sided two-way ranging} $\Rightarrow$ precisão de \textbf{10--30~cm};
        \item robusto a multipercurso (usa o primeiro caminho, não o RSSI).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Aplicações}
      \begin{itemize}
        \item \textbf{chaves digitais} (carro/casa): acesso passivo e anti-relé;
        \item localização \emph{indoor} (logística, hospitais, estádios);
        \item pagamento por aproximação seguro;
        \item complementa o BLE: BLE descobre, UWB mede com precisão.
      \end{itemize}
    \end{column}
  \end{columns}
  \begin{alertblock}{Por que UWB?}
    Enquanto o RSSI do BLE erra metros, o tempo de voo do UWB acerta \textbf{centímetros} --- essencial para \emph{car key} e navegação indoor.
  \end{alertblock}
\end{frame}

\section{LPWAN: LoRa e Sigfox}

\begin{frame}{LPWAN para IoT: LoRa vs.\ Sigfox}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{LoRa / LoRaWAN}
      \begin{itemize}
        \item modulação \textbf{Chirp Spread Spectrum (CSS)}
        \item banda ISM sub-GHz (868~MHz EU, 915~MHz US/BR)
        \item taxas 0,3--50~kbps; alcance 2--15~km
        \item topologia \emph{star-of-stars}: sensores $\rightarrow$ gateways $\rightarrow$ servidor
        \item ALOHA puro + canais com \emph{duty cycle}
        \item autenticação AES-128 (join E2E)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Sigfox}
      \begin{itemize}
        \item banda ultra-estreita (UNB, 100~Hz)
        \item uplink 100 bps; 140 mensagens/dia
        \item alcance até 50~km (rural)
        \item rede proprietária (operador regional)
      \end{itemize}
      \pause
      \begin{block}{Comparação prática}
        LoRaWAN: mais flexível, redes privadas, bidirecional.
        Sigfox: ultra-simples, muito baixo consumo, unidirecional.
      \end{block}
    \end{column}
  \end{columns}
  \note{Relacionar SF e tempo no ar: SF maior = símbolos mais longos, maior sensibilidade e alcance, porém menor taxa e maior consumo — base para o exercício resolvido.}
\end{frame}

\begin{frame}{Arquitetura LoRaWAN: star-of-stars}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{Elementos}
      \begin{itemize}
        \item \textbf{End devices}: sensores com rádio LoRa (CSS), acesso ALOHA puro;
        \item \textbf{Gateways} (concentradores): rádio $\leftrightarrow$ IP (backhaul Ethernet/4G), ponte transparente;
        \item \textbf{Network Server (NS)}: deduplicação, segurança (AES-128), \textbf{ADR} e agendamento de downlink;
        \item \textbf{Application Server (AS)}: recebe e entrega os payloads à aplicação;
        \item \textbf{Join Server (JS)}: ativação OTAA (chaves de sessão).
      \end{itemize}
    \end{column}

\begin{frame}{Sigfox em profundidade}
  \begin{itemize}
    \item \textbf{UNB}: sinais de $\sim$100~Hz na banda ISM sub-GHz; uplink \textbf{100~bps} (downlink 600~bps);
    \item limites: \textbf{140 uplink/dia} e \textbf{4 downlink/dia}, payload de até 12 (ul) / 8 (dl) bytes;
    \item \textbf{downlink condicional}: resposta só após um uplink (2 janelas de recepção);
    \item \textbf{geolocalização}: triangulação por \textbf{RSSI} entre base stations e \textbf{TDOA} --- sem GPS no dispositivo;
    \item \textbf{modelo de operadora}: rede regional operada (B2B2C); o dispositivo assina um plano (``SIM'' de rádio), roaming por acordos;
    \item alcance até \textbf{50~km} rural / 10~km urbano; boa penetração em subsolo (medidores).
  \end{itemize}
\end{frame}

\begin{frame}{Starlink: arquitetura em detalhe}
  \begin{columns}

      \textbf{Limitações (honestas)}
      \begin{itemize}
        \item \textbf{chuva/neve} atenuam as bandas Ka/Ku (dezenas de dB em tempestades);
        \item interferência na \textbf{astronomia}: brilho e rastros nos telescópios;
        \item \textbf{lixo espacial} e risco de colisão em LEO;

  \end{columns}
\end{frame}

\begin{frame}{Rádio cognitivo: ciclo sense--decide--act}

        \item proteção do \emph{broadcast}: \textbf{geolocation database} + sensoriamento;

\begin{frame}{Exercício resolvido: LoRaWAN --- tempo no ar}
  \small
  \textbf{Dados}: $SF=12$, $BW=125$~kHz, $CR=4/5$, payload $PL=20$ bytes, header explícito ($H=0$), CRC ativado, $DE=0$.
  \begin{block}{Passo 1 --- Símbolo e preâmbulo}
    $T_{sym}=2^{SF}/BW=4096/125\,000=32{,}768$~ms; \quad $T_{preamble}=12{,}25\cdot T_{sym}\approx 401{,}4$~ms
  \end{block}
  \pause
  \begin{block}{Passo 2 --- Símbolos de payload}
    $n_{payload}=8+\big\lceil(8PL-4SF+28+16\,CRC-20H)/(4(SF-2DE))\big\rceil\cdot(CR+4)$\\
    $\phantom{n_{payload}}=8+\lceil 156/48\rceil\cdot 8=8+4\cdot 8=40$ símbolos
  \end{block}
  \pause
  \begin{block}{Passo 3 --- Tempo no ar}
    $T_{payload}=40\cdot 32{,}768\approx 1310{,}7$~ms \quad $\Rightarrow$ \quad $T_{total}=T_{preamble}+T_{payload}\approx \mathbf{1{,}71}$~\textbf{s}
  \end{block}
  \pause
  \begin{alertblock}{Interpretação}
    Com \emph{duty cycle} de 1\% (EU868), entre duas transmissões deve haver $\geq 100\times 1{,}71\approx 171$~s ($\approx 2{,}9$~min).
  \end{alertblock}
\end{frame}

\begin{frame}{Exercício resolvido: LoRaWAN --- alcance por orçamento de enlace}
  \textbf{Dados}: $P_{TX}=14$~dBm, sensibilidade $S_{RX}=-137$~dBm (SF12/BW125), frequência $868$~MHz.
  \begin{block}{Orçamento de enlace}
    \begin{itemize}
      \item perda máxima admissível: $L_{max}=P_{TX}-S_{RX}=14-(-137)=\mathbf{151}$~\textbf{dB}
      \item perda no espaço livre (modelo): $PL(d)=32{,}44+20\log_{10}(f[\text{MHz}])+20\log_{10}(d[\text{km}])$
      \item a $10$~km: $PL=32{,}44+20\log_{10}(868)+20\log_{10}(10)=32{,}44+58{,}77+20\approx 111$~dB
    \end{itemize}
  \end{block}
  \pause
  \begin{exampleblock}{Resultado}
    Margem de $151-111\approx 40$~dB a $10$~km: o link é viável mesmo com perdas por obstáculos/multipercurso; na prática, o alcance LoRaWAN fica entre 2--15~km (rural até $\sim$50~km no Sigfox).
  \end{exampleblock}
\end{frame}

\begin{frame}{Resumo da aula}
  \begin{itemize}
    \item \textbf{Wi-Fi 6/7}: OFDMA, MU-MIMO, QAM-1024/4096, canais de 320~MHz e \emph{multi-link operation} (MLO);
    \item \textbf{LPWAN}: LoRaWAN (CSS, \emph{star-of-stars}, \emph{duty cycle}) e Sigfox (UNB) para IoT de baixo custo e consumo; NB-IoT/LTE-M como alternativa celular com QoS;
    \item \textbf{5G NR}: casos eMBB, uRLLC e mMTC viabilizados por \emph{beamforming} massivo, \emph{network slicing} e MEC;
    \item \textbf{Satélites LEO (Starlink)}: latência 20--40~ms, enlaces a laser ISL e integração NTN (3GPP Rel.\ 17+);
    \pause
    \item \textbf{Arquiteturas emergentes}: DTN (Bundle Protocol), FANET/VANET, SDR e SDN como objetos de pesquisa ativa.
  \end{itemize}
\end{frame}
