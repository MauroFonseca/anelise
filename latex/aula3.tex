\section{Características e objetivos}

\begin{frame}{Características das redes sem fio (recursos)}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Recursos limitados}
      \begin{itemize}
        \item quantidade de energia limitada
        \item capacidade de processamento limitada (PDAs, sensores,\ldots)
        \item alcance de transmissão pequeno
      \end{itemize}
      \textbf{Topologia dinâmica}
      \begin{itemize}
        \item mobilidade
        \item falha de nós
        \item nós inativos em períodos de baixa atividade
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Escalabilidade}
      \begin{itemize}
        \item necessária redundância --- nós podem falhar
      \end{itemize}
      \textbf{Tempo de vida da rede}
      \begin{itemize}
        \item deve ser o maior possível (principalmente nós sensores)
      \end{itemize}
      \begin{alertblock}{Consequência}
        O roteamento deve ser \textbf{eficiente em energia} e \textbf{adaptativo}.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Objetivos do roteamento ad hoc}
  \begin{block}{Objetivo 1: conectividade total}
    Desenvolver uma infraestrutura de rede onde necessário:
    \begin{itemize}
      \item roteamento ad hoc $\rightarrow$ redes auto-organizáveis, sem manutenção nem administração
      \item roteamento passando por outros terminais
    \end{itemize}
  \end{block}
  \begin{block}{Objetivo 2: mesmo serviço que uma rede com infraestrutura}
    \begin{itemize}
      \item mobilidade, descoberta de serviço, roteamento ad hoc
      \item outros: autoconfiguração, localização, segurança, consciência do papel, redes de sensores
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Arquitetura ad hoc}
  \begin{itemize}
    \item criação de uma \textbf{conectividade multisalto} entre um grupo de nós sem fio, frequentemente em movimento
    \item os nós móveis atuam como \textbf{roteadores} e como \textbf{origem/destino}
    \item necessidade de um protocolo de roteamento que encontre uma ou várias rotas multisalto
      \begin{itemize}
        \item dinâmico para se adaptar à topologia cambiante
        \item desafios: interferências, limitações de potência
      \end{itemize}
  \end{itemize}

\begin{frame}{Necessidades do roteamento ad hoc}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{Distribuição dos caminhos}
      \begin{itemize}
        \item caminhos multisalto
        \item sem loops (bucles)
        \item minimizar \emph{overhead}
        \item multicast? inundação eficiente?
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{itemize}
        \item \textbf{adaptativa à topologia dinâmica}
        \item \textbf{baixo consumo}: banda e potência
        \item \textbf{escalabilidade} (número de nós)
        \item efeitos diante da queda de enlaces
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Problemas dos protocolos clássicos DV/LS}
  \begin{block}{DV --- Distance Vector (ex.: RIP)}
    \begin{itemize}
      \item podem formar \textbf{bucles (loops)}: muito desperdício em meios sem fio (banda, potência)
      \item normalmente complexo evitá-los
    \end{itemize}
  \end{block}
  \begin{block}{LS --- Link State (ex.: OSPF)}
    \begin{itemize}
      \item custo para armazenar tabelas e overhead de comunicação
    \end{itemize}
  \end{block}
  \begin{alertblock}{Ainda mais problemas}
    \begin{itemize}
      \item atualização periódica gasta energia (emissor e receptor)
      \item a convergência deve ser \textbf{rápida em ad hoc}, sem atualizações muito frequentes
    \end{itemize}
  \end{alertblock}
\end{frame}

\section{AODV}

\begin{frame}{AODV --- Ad hoc On-Demand Distance Vector}
  \begin{itemize}
    \item Charles Perkins et al. (Nokia Research Center)
    \item \textbf{Protocolo adaptado a redes com alterações frequentes de rotas} (manutenção de rotas improdutiva)
    \item pode se adaptar a redes com grande taxa de transmissão
    \item baseado no \textbf{roteamento de vetor de distância sob demanda}
      \begin{itemize}
        \item descoberta de rota por \emph{inundação} (flooding)
        \item cálculo de tabelas de roteamento (armazenamento temporário)
        \item avaliação de overheads
      \end{itemize}
  \end{itemize}
  \begin{exampleblock}{RFC}
    RFC 3561 (2003) --- \emph{Ad hoc On-Demand Distance Vector (AODV) Routing}.
  \end{exampleblock}
\end{frame}

\begin{frame}{Mensagens do protocolo AODV}
  \begin{description}
    \item[Route Request (RREQ)] ``Preciso de uma rota''
    \item[Route Response (RREP)] ``Anunciar a rota''
    \item[Route Error (RERR)] ``Anular a rota''
  \end{description}
  \begin{itemize}
    \item respostas periódicas da rota aos vizinhos que atuam como \texttt{HELLO}, para instalação e renovação
  \end{itemize}

\begin{frame}{Descoberta da rota (AODV)}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{enumerate}
        \item Uma rota entre 2 nós é encontrada mediante o envio de \textbf{RREQ} a toda a localidade
        \item inicia em localidade pequena; aumenta com falhas na descoberta
        \item um \textbf{RREP} é enviado:
          \begin{itemize}
            \item pelo destino
            \item por um nó vizinho que já tenha rota estabelecida
          \end{itemize}
        \item os nós intermediários armazenam temporariamente a rota
      \end{enumerate}
      \textbf{Erros de rota}
      \begin{itemize}
        \item a rota expira se não se atualiza
        \item os roteadores armazenam os usuários da rota; ao expirar notificam-se (RERR)
        \item nova mensagem \texttt{route request} será enviada
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{center}

\section{DSR}

\begin{frame}{DSR --- Dynamic Source Routing}
  \begin{itemize}
    \item David Johnson et al. (Rice University)
    \item \textbf{Mensagens:} Route Discovery, Route Maintenance
    \item \textbf{Múltiplas rotas} disponíveis
    \item permite a cada emissor \textbf{selecionar e controlar as rotas utilizadas}
    \item suporta mais de 200 nós
    \item alta mobilidade dos nós
  \end{itemize}
  \begin{exampleblock}{RFC}
    RFC 4728 (2007) --- \emph{The Dynamic Source Routing Protocol for Mobile Ad Hoc Networks (IPv4)}.
  \end{exampleblock}
\end{frame}

\begin{frame}{Descoberta da rota (DSR)}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{enumerate}
        \item Uma rota entre 2 nós é encontrada mediante \textbf{RREQ}
        \item o RREQ \textbf{constrói uma rota para a fonte} em cada caminho (nós registram o seu predecessor)
        \item o \textbf{primeiro RREQ que chega é aceito}; o destino responde por esse caminho informando a rota
        \item o roteamento \textbf{baseado na fonte} é utilizado pelo tráfego de dados
      \end{enumerate}
      \begin{exampleblock}{Exemplo}
        Rota detectada: \texttt{(A, B, F, J, K)} --- incluída no cabeçalho do pacote.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.5\textwidth}

\begin{frame}{Confiabilidade do enlace e manutenção (DSR)}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{Confiabilidade do enlace}
      \begin{itemize}
        \item rotas de trânsito armazenadas (para saber a quem enviar RERR)
        \item confirmações da camada de rede avaliadas (transferência, detecção de nós com falhas)
        \item após múltiplas retransmissões:
          \begin{itemize}
            \item o nó declarado fora de serviço
            \item RERR aos emissores que usam esse nó
            \item os pacotes de dados são rejeitados
          \end{itemize}
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Manutenção da rota}
      \begin{itemize}
        \item se um nó roteador ou enlace muda, as rotas baseadas na fonte deixam de ser válidas
        \item os nós respondem com \textbf{RERR}, ativando a descoberta de nova rota
        \item os nós podem tentar alterar a rota e difundi-la
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{TBRPF}

\begin{frame}{TBRPF --- Topology Broadcast Based on Reverse-Path Forwarding}
  \begin{itemize}
    \item Richard Ogier et al. (SRI International)
    \item os nós enviam \textbf{só uma parte da tabela de roteamento aos vizinhos}
    \item descoberta de vizinhança com \textbf{HELLO diferenciados}: só se informam mudanças de estado dos vizinhos
    \item utiliza o algoritmo \textbf{Shortest Path First} (menor caminho) para seleção da melhor rota
  \end{itemize}
  \begin{exampleblock}{RFC}
    RFC 3684 (2004) --- \emph{Topology Dissemination Based on Reverse-Path Forwarding (TBRPF)}.
  \end{exampleblock}
  \begin{block}{Mensagens}
    \begin{itemize}
      \item \texttt{HELLO}: descoberta e manutenção da relação entre vizinhos
      \item \texttt{TOPOLOGY UPDATE}: distribuição de um subconjunto de informação para interconectividade
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Distribuição da atualização da topologia (TBRPF)}
  \begin{itemize}
    \item informação da topologia é distribuída ao subconjunto de nós que possam precisar dela
      \begin{itemize}
        \item é necessária informação sobre um nó se o roteamento o utiliza\ldots
      \end{itemize}
    \item por isso um nó roteador calcula todas as rotas de todos os nós que possuam tal informação
  \end{itemize}

\section{OLSR}

\begin{frame}{OLSR --- Optimized Link State Routing}
  \begin{itemize}
    \item Philippe Jacquet et al. (INRIA Rocquencourt)
    \item otimização para redes ad hoc mediante \textbf{MPRs (Multipoint Relays)}
    \item só os nós MPR podem transmitir as mensagens de descoberta da rede
    \item adaptado a redes com \textbf{baixa mobilidade}
    \item utiliza \textbf{Shortest Path First} (menor caminho)
  \end{itemize}
  \begin{exampleblock}{RFC}
    RFC 3626 (2003) --- \emph{Optimized Link State Routing Protocol (OLSR)}.
  \end{exampleblock}
  \begin{block}{Mensagens}
    \begin{itemize}
      \item \texttt{HELLO}: descoberta e manutenção de vizinhos
      \item \texttt{TC} (Topology Control): distribuição de informação de conectividade
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Relações entre nós vizinhos e seleção de MPR}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Cada equipamento emite um \texttt{HELLO} periódico}
      \begin{itemize}
        \item anuncia aos vizinhos
        \item determina quem ainda está conectado
        \item seleciona os nós MPR (MultiPoint Relays)
      \end{itemize}
      \textbf{MPRs}
      \begin{itemize}
        \item responsáveis por transmitir a informação da topologia
        \item atuam como roteadores entre nós destino
        \item minimizam a retransmissão da informação
        \item formam um \textbf{backbone de roteamento}
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}

\begin{frame}{Comparação: proativos vs.\ reativos}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{Reativos (sob demanda) --- AODV, DSR}
      \begin{itemize}
        \item baseados em inundação
        \item não há garantia de rota ótima
        \item menor overhead quando há pouco tráfego
        \item atraso (latência) na descoberta de rota
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Proativos --- OLSR, TBRPF}
      \begin{itemize}
        \item descobrem continuamente a topologia
        \item mensagens HELLO periódicos
        \item garantia de rota ótima
        \item possibilidade de obter várias rotas
        \item maior overhead contínuo
      \end{itemize}
    \end{column}
  \end{columns}

\section{Roteamento em redes de sensores}

\begin{frame}{Roteamento em redes de sensores: características}

        \item energia limitada (baterias)
        \item capacidade de processamento limitada
        \item alcance pequeno

      \begin{itemize}

        \item sensores inativos em períodos de baixa atividade
      \end{itemize}

    \begin{column}{0.5\textwidth}
      \textbf{Escalabilidade} e \textbf{tempo de vida}
      \begin{itemize}
        \item redundância --- nós podem falhar
        \item o objetivo principal é \textbf{economizar energia}
      \end{itemize}
      \textbf{Tipos de roteamento}
      \begin{itemize}
        \item plano (flat), comunicação direta
        \item multisalto de energia mínima
        \item clusterização (agrupamento), clusterização dinâmica
        \item inundação (flooding), gradiente
        \item roteamento geográfico
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Difusão Direcionada (Directed Diffusion)}
  \begin{itemize}
    \item proposto por Estrin et al. (1999)
    \item o nó \textbf{nomeia os dados} usando um ou mais atributos
    \item o nó cliente \textbf{requer os dados} mediante \textbf{interesses}
    \item nós intermediários \textbf{propagam os interesses}
    \item os interesses estabelecem \textbf{gradientes de dados} em direção ao nó que expressou o interesse
  \end{itemize}

\begin{frame}{LEACH --- Low-Energy Adaptive Clustering Hierarchy}
  \begin{itemize}
    \item proposto por Heinzelman et al. (2000)
    \item coleta de dados \textbf{centralizada e periódica} (redes proativas)
    \item os nós se organizam em \textbf{aglomerados (clusters)}:
      \begin{itemize}
        \item cada cluster tem um \textbf{cluster-head (base local)}
        \item cada nó decide a sua base local (menor custo de comunicação)
        \item \textbf{rotação aleatória de bases} para nós com maior energia
        \item \textbf{agregação de dados local}
      \end{itemize}
    \item cada base cria um \textbf{agendamento TDMA} (rádio desligado, sem colisões)
    \item cada cluster usa um código CDMA diferente
    \item alto consumo nas bases, mas só 5\% são bases; rotação aleatória $\rightarrow$ \textbf{maior vida da rede}
  \end{itemize}
  \note{Enfatize que a rotação aleatória do papel de base é o mecanismo que distribui o consumo de energia entre os nós; apenas 5\% dos nós atuam como base por vez.}
\end{frame}

\begin{frame}{Comparação: roteamento hierárquico vs.\ plano}
  \begin{center}
    \scriptsize
    \begin{tabular}{@{}p{6.4cm}p{6.4cm}@{}}
      \toprule
      \textbf{Hierárquico} & \textbf{Plano}\\
      \midrule
      evita colisões (TDMA) & não evita colisões\\
      agregação nos cluster-heads & agregação nos nós intermediários\\
      requer sincronização local e global & enlaces formados ``on-the-fly''\\
      overhead de formação de cluster & rotas só nas regiões com dados\\
      dissipação uniforme de energia & dissipação depende do tráfego\\
      alocação justa de canal & justiça não garantida\\
      baixa latência (cluster-heads disponíveis) & latência ao esperar nós intermediários\\
      roteamento simples, não ótimo & roteamento complexo, ótimo\\
      \bottomrule
    \end{tabular}
  \end{center}
\end{frame}

\section{Exercícios e avaliação}

\begin{frame}{Exercício resolvido: overhead de inundação \emph{vs.} MPR}
  \begin{block}{Enunciado}
    Considere uma MANET com $N = 9$ nós organizados em uma malha $3\times 3$. O nó de origem S deseja descobrir uma rota até o nó D, transmitindo um \textbf{RREQ} (AODV) que é retransmitido por inundação.
  \end{block}
  \begin{block}{Pergunta 1 --- inundação simples}
    Se cada nó retransmite o RREQ recebido exatamente uma vez, quantas retransmissões ocorrem?
    \[
      R_{\text{flood}} = N - 1 = 9 - 1 = 8\ \text{retransmissões}
    \]
    (além da transmissão inicial do nó S, totalizando 9 transmissões na rede).
  \end{block}
  \begin{block}{Pergunta 2 --- com MPRs no OLSR}
    Suponha que apenas $M = 2$ nós sejam selecionados como MPR. Quantas retransmissões são feitas e qual a redução de overhead?
    \[
      R_{\text{MPR}} = M = 2, \qquad
      \Delta = \frac{R_{\text{flood}} - R_{\text{MPR}}}{R_{\text{flood}}} \times 100\%
            = \frac{8 - 2}{8} \times 100\% = 75\%
    \]
  \end{block}
  \begin{alertblock}{Resultado}
    As retransmissões caem de 8 para 2, uma redução de \textbf{75\%} no overhead de inundação com o uso de MPRs.
  \end{alertblock}
\end{frame}

\begin{frame}{Resumo da aula}
  \begin{itemize}
    \item Redes ad hoc: sem infraestrutura; os nós atuam como roteadores; topologia dinâmica e energia limitada.
    \item Dois grupos de protocolos: \textbf{proativos} (OLSR, TBRPF) e \textbf{reativos} (AODV, DSR).
    \item \textbf{AODV}: descoberta de rota sob demanda com \textbf{RREQ/RREP} e aviso de erro com \textbf{RERR}; tabelas nos nós intermediários.
    \item \textbf{DSR}: roteamento \textbf{na fonte}; a rota completa vai no cabeçalho do pacote; suporta múltiplas rotas.
    \item \textbf{OLSR}: usa \textbf{MPRs} para reduzir retransmissões; \textbf{TBRPF}: difunde apenas parte da topologia.
    \item Redes de sensores: roteamento \textbf{plano} (ex.: Directed Diffusion) \emph{vs.} \textbf{hierárquico} (ex.: LEACH); o objetivo é \textbf{economizar energia}.
    \item Redes mesh: protocolos ad hoc com métricas adaptadas (ex.: Roofnet, GT-Mesh).
  \end{itemize}
\end{frame}

\begin{frame}{Autoavaliação}
  \begin{enumerate}
    \item Sobre a classificação dos protocolos de roteamento ad hoc, é correto afirmar:
      \begin{itemize}
        \item (a) AODV e DSR são proativos, pois mantêm rotas sempre disponíveis.
        \item (b) OLSR e TBRPF são reativos, pois descobrem rotas sob demanda.
        \item (c) AODV e DSR são reativos, pois descobrem a rota somente quando há tráfego.
      \end{itemize}
      \pause
      \textbf{Gabarito:} (c)
    \item No AODV, a descoberta de rota é iniciada:
      \begin{itemize}
        \item (a) pelo envio de \textbf{RREP} pelo destino.
        \item (b) pelo envio de \textbf{RREQ} por inundação a partir da origem.
        \item (c) pelo envio de \textbf{RERR} aos vizinhos.
      \end{itemize}
      \pause
      \textbf{Gabarito:} (b)
    \item No OLSR, o objetivo dos MPRs (Multipoint Relays) é:
      \begin{itemize}
        \item (a) aumentar o overhead de controle.
        \item (b) eliminar a necessidade de mensagens \texttt{HELLO}.
        \item (c) reduzir o número de retransmissões na inundação de controle.
      \end{itemize}
      \pause
      \textbf{Gabarito:} (c)
    \item No LEACH, a rotação aleatória do papel de cluster-head (base local) tem por finalidade:
      \begin{itemize}
        \item (a) distribuir o consumo de energia entre os nós e aumentar o tempo de vida da rede.
        \item (b) concentrar o consumo de energia em um único nó.
        \item (c) eliminar a necessidade de agendamento TDMA.
      \end{itemize}
      \pause
      \textbf{Gabarito:} (a)
  \end{enumerate}
\end{frame}

\begin{frame}{Laboratório sugerido: simulação de MANET no ns-3}
  \begin{block}{Objetivo de aprendizagem}
    Verificar empiricamente o \textbf{trade-off} entre protocolos reativos (AODV) e proativos (OLSR): overhead de descoberta de rota \emph{vs.} latência de entrega.
  \end{block}
  \begin{block}{Passos}
    \begin{enumerate}
      \item Criar uma simulação no ns-3 com $N = 20$ nós (módulo \texttt{wifi-adhoc}) em uma área de $500\times 500$\,m, com mobilidade \texttt{RandomWaypointMobilityModel}.
      \item Configurar o canal 802.11a (\texttt{WifiHelper}) e instalar o roteamento \texttt{ns3::AodvRoutingProtocol} (\texttt{AodvRoutingHelper}).
      \item Gerar tráfego UDP entre um par origem--destino com \texttt{UdpClientServerHelper} (ou \texttt{OnOffApplication}).
      \item Medir com \texttt{FlowMonitor}: taxa de entrega de pacotes, atraso fim a fim e overhead de roteamento (mensagens RREQ/RREP emitidas, via \texttt{TraceSource}).
      \item Repetir variando o número de nós (10, 20, 50) ou a velocidade (2, 10, 20\,m/s) e comparar com o OLSR (\texttt{OlsrRoutingHelper}).
    \end{enumerate}
  \end{block}
\end{frame}
