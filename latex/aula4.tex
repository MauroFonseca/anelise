\section{Enlaces e protocolos de acceso múltiple}

\begin{frame}{Enlaces de acceso múltiple e protocolos}
  \textbf{Dous tipos de enlaces:}
  \begin{itemize}
    \item \textbf{Punto-a-punto} (ex.: protocolo PPP)
    \item \textbf{Broadcast} (fio ou medio compartilhado): Ethernet tradicional, 802.11 LAN sem fio
  \end{itemize}
  \begin{block}{Problema central}
    Canal de comunicação único e compartilhado:
    \begin{itemize}
      \item dúas ou mais transmissões simultáneas $\rightarrow$ \textbf{interferência}
      \item \textbf{colisión} se un nó recibe dúas ou mais sinais ao mesmo tempo
      \item protocolo de acceso múltiple $=$ algoritmo distribuído que determina quando cada estação pode transmitir
      \item \alert{não há canal out-of-band para coordinación}
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Protocolo ideal de acceso múltiple}
  Canal de broadcast de taxa $R$ bps. Obxectivos:
  \begin{enumerate}
    \item Quando un nó quere transmitir, pode envialo a taxa $R$
    \item Quando $M$ nós queren transmitir, cada un envía a taxa media $R/M$
    \item \textbf{Totalmente descentralizado:}
      \begin{itemize}
        \item ningún nó especial para coordinar
        \item ningunha sincronização de reloxos nin compartimentos
      \end{itemize}
    \item \textbf{Simples}
  \end{enumerate}
  \begin{alertblock}{Trade-off}
    Particionamento (obxectivos 1--2) vs.\ acceso aleatorio (obxectivos 3--4): não se podem ter todos à vez $\rightarrow$ por iso existen distintas familias de protocolos.
  \end{alertblock}
\end{frame}

\section{TDMA e FDMA}

\begin{frame}{TDMA --- Time Division Multiple Access}
  \begin{itemize}
    \item acceso ao canal por ``turnos''
    \item cada estação controla un \textbf{slot} (compartimento) de tamaño fixo (tempo de transmissão dun pacote) em cada turno
    \item divide o tempo em \textbf{quadros temporais} com $N$ slots ($N=$ número de computadores)
    \item \alert{slots não usados desperdícianse}
  \end{itemize}

\section{Acceso aleatorio: ALOHA}

\begin{frame}{Protocolos de acceso aleatorio}
  \begin{block}{Ideas base}
    \begin{itemize}
      \item quando o nó ten un pacote: transmite a toda a taxa $R$
      \item não há regra de coordinación a priori
      \item dúas ou mais transmissões simultáneas $\rightarrow$ \textbf{colisión}
    \end{itemize}
  \end{block}
  \begin{block}{O protocolo especifica}
    \begin{itemize}
      \item como detectar colisións
      \item como recuperarse (retransmisións atrasadas)
    \end{itemize}
  \end{block}
  \textbf{Exemplos:} slotted ALOHA, ALOHA puro, CSMA, CSMA/CD, CSMA/CA.
\end{frame}

\begin{frame}{Slotted ALOHA: operación}
  \textbf{Suposicións:}
  \begin{itemize}
    \item todos os frames do mesmo tamaño
    \item tempo dividido em slots (tempo de transmissão dun frame)
    \item os nós transmiten só ao inicio de slot; están sincronizados
    \item se 2 ou mais transmiten no slot, todos detectan a colisión
  \end{itemize}
  \textbf{Operación:}
  \begin{itemize}
    \item quando un nó obtén un novo frame, transmite no próximo slot
    \item sem colisión: pode enviar o novo frame no próximo slot
    \item com colisión: retransmite em cada slot subsecuente com probabilidade $p$ ata éxito
  \end{itemize}
  \begin{center}
    \begin{tikzpicture}[font=\scriptsize,
        slot/.style={draw,minimum width=1cm,minimum height=0.65cm,anchor=south west,inner sep=0.5mm},
      ]

      \node[slot,fill=aneliseLaranja!40,right=0.1cm of s1](s2){};
      \node[slot,fill=aneliseGris!15,right=0.1cm of s2](s3){};
      \node[slot,fill=aneliseVerde!45,right=0.1cm of s3](s4){};
      \node[slot,fill=aneliseGris!15,right=0.1cm of s4](s5){};
      \node[below=1mm of s0,font=\tiny]{C};

      \node[below=1mm of s5,font=\tiny]{E};
    \end{tikzpicture}
  \end{center}
  \begin{center}
    \scriptsize C = colisión, E = slot baleiro, S = éxito
  \end{center}
\end{frame}

\begin{frame}{Eficiencia do Slotted ALOHA}
  \begin{itemize}
    \item eficiencia $=$ fracción de slots com éxito quando hai muitos nós com muitos frames
    \item $N$ nós com muitos frames; cada un transmite no slot com probabilidade $p$
    \item prob.\ de éxito dun nó $= p(1-p)^{N-1}$
    \item prob.\ de éxito de calquera nó $= Np(1-p)^{N-1}$
    \item óptimo $p^*$ maximiza $Np(1-p)^{N-1}$
    \item límite quando $N\to\infty$: $\dfrac{1}{e} = 0{,}37$
  \end{itemize}
  \begin{alertblock}{Conclusión}
    No máximo: uso do canal para datos útiles: $\mathbf{37\%}$ do tempo!
  \end{alertblock}
\end{frame}

\begin{frame}{ALOHA puro (unslotted)}
  \begin{itemize}
    \item operación mais simple: \textbf{sem sincronização}
    \item o pacote envíase sem esperar inicio de slot
    \item a probabilidade de colisión aumenta:
      \begin{itemize}
        \item pacote enviado em $t_0$ colide com outros enviados em $[t_0-1,\ t_0+1]$
      \end{itemize}
  \end{itemize}
  \begin{block}{Eficiencia}
    \begin{align*}
      P(\text{éxito dun nó}) &= p\,(1-p)^{N-1}\,(1-p)^{N-1} = p(1-p)^{2(N-1)}
    \end{align*}
    co $p$ óptimo e $n\to\infty$: $\dfrac{1}{2e} = 0{,}18$
  \end{block}
  \begin{alertblock}{Conclusión}
    No máximo: $\mathbf{18\%}$ do tempo. Pior que slotted ALOHA.
  \end{alertblock}
\end{frame}

\section{CSMA, CSMA/CD}

\begin{frame}{CSMA --- Carrier Sense Multiple Access}
  \begin{block}{Idea}
    \textbf{Escoita antes de transmitir:}
    \begin{itemize}
      \item se o canal parece baleiro $\rightarrow$ transmite o pacote
      \item se o canal está ocupado $\rightarrow$ adía a transmissão
    \end{itemize}
    \emph{``Non interrumpas aos demáis!''}
  \end{block}
  \begin{block}{Colisións no CSMA}
    \begin{itemize}
      \item o atraso de propagación implica que dous nós podem não ouvir a transmissão do outro
      \item colisión: todo o tempo de transmissão do pacote desperdíciase
      \item \textbf{importancia da distancia e do atraso de propagación}
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{CSMA/CD --- Collision Detection}
  \begin{itemize}
    \item detección de portadora e deferencia como CSMA
    \item colisións detectadas em tempo mais curto
    \item transmissões com colisión interrómpense $\rightarrow$ menos desperdicio
  \end{itemize}
  \begin{block}{Como se detecta a colisión?}
    \begin{itemize}
      \item \textbf{fácil em LANs cableadas}: medición da intensidade do sinal, comparación do transmitido e recibido
      \item \textbf{difícil em LANs sem fio}: o receptor está apagado mentres transmite!
    \end{itemize}
  \end{block}
  \begin{exampleblock}{Ethernet usa CSMA/CD}
    \begin{itemize}
      \item sem slots; \emph{carrier sense}: não transmite se detecta outra estação
      \item \emph{collision detection}: aborta ao detectar outra transmissão
      \item \emph{random access}: antes de retransmitir espera un período aleatorio
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{Algoritmo CSMA/CD da Ethernet}
  \begin{enumerate}
    \item A estação recibe un datagrama da capa de rede e crea un frame
    \item Se detecta canal libre, comeza a transmitir; se ocupado, espera ata que quede libre
    \item Se transmite o frame completo sem detectar outra transmissão $\rightarrow$ ``éxito!''
    \item Se detecta outra transmissão mentres transmite $\rightarrow$ aborta e envía \emph{jam signal} (48 bits)
    \item Tras abortar: \emph{exponential backoff}: após a $m$-ésima colisión, seleciona $K\in\{0,1,\ldots,2^m-1\}$, espera $K\cdot 512$ bit-times e volve ao paso 2
  \end{enumerate}
  \begin{exampleblock}{Valores típicos}
    Bit time (10~Mbps) $= 0{,}1\,\mu$s; $K=1023 \Rightarrow$ espera $\approx 50$~ms.
    Backoff: primeira colisión $K\in\{0,1\}$; segunda $\{0,1,2,3\}$; após 10 ou mais $\{0,\ldots,1023\}$.
  \end{exampleblock}
\end{frame}

\begin{frame}{Eficiencia do CSMA/CD}
  \begin{itemize}
    \item $t_{\text{prop}}$ = propagación máxima entre 2 nós na LAN
    \item $t_{\text{trans}}$ = tempo de transmissão dun frame de tamaño máximo
  \end{itemize}
  \begin{block}{Ethernet 10~Mbps}
    $t_{\text{prop}} = 51{,}2\,\mu$s, $t_{\text{trans}} = 1{,}2$~ms $\Rightarrow$ eficiencia $\approx 82{,}6\%$.
  \end{block}
  \begin{itemize}
    \item tende a 1 quando $t_{\text{prop}}\to 0$ ou $t_{\text{trans}}\to\infty$
    \item \textbf{muito melhor que ALOHA}, e segue sendo descentralizado, simple e barato
  \end{itemize}
  \begin{alertblock}{Nota actualizada}
    Ethernet moderno (switches) é full-duplex; CSMA/CD queda como material histórico.
  \end{alertblock}
\end{frame}

\section{Pasaxe de turno}

\begin{frame}{Protocolos MAC com pasaxe de turno}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Por que outra familia?}
      \begin{itemize}
        \item particionamento: eficiente com carga alta, ineficiente com carga baixa (un nó só recibe $1/N$)
        \item acceso aleatorio: eficiente com carga baixa (un nó usa todo), colisións com carga alta
        \item \textbf{pasaxe de turno}: busca o melhor de ambos
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Polling}
      \begin{itemize}
        \item nó-mestre ``invita'' aos escravos a transmitir un a un
        \item problemas: overhead de polling, latencia, punto único de falla (mestre)
      \end{itemize}
      \textbf{Token passing}
      \begin{itemize}
        \item un token pasa de nó a nó secuencialmente (mensagem token)
        \item problemas: overhead de token, latencia, punto único de falla (token)
      \end{itemize}
    \end{column}
  \end{columns}

\section{CSMA/CA e 802.11}

\begin{frame}{Wireless Ethernet: por que não CSMA/CD?}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{802.3 (Ethernet)}: CSMA/CD
      \begin{itemize}
        \item Carrier Sense + Collision Detect
      \end{itemize}
      \textbf{802.11 (WLAN)}: CSMA/CA (Collision Avoidance)
      \begin{itemize}
        \item tanto A como C ``senten'' o canal desocupado ao mesmo tempo e ambos envían
        \item co CSMA/CD a colisión pode detectarse no emisor; \alert{por que não co CSMA/CA?}
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Razóns}
        \begin{itemize}
          \item comunicação \textbf{half-duplex}
          \item grande diferenza entre potência de transmissão e recepção
          \item medio sem fio
        \end{itemize}
      \end{block}
      \begin{center}

\end{frame}

\begin{frame}{IEEE 802.11 MAC: CSMA/CA}
  \begin{block}{Emisor 802.11 CSMA}
    \begin{itemize}
      \item se o canal é sentido baleiro por \textbf{DIFS} segundos $\rightarrow$ envía o frame completo (sem detección de colisión)
      \item se o canal é sentido ocupado $\rightarrow$ \textbf{binary backoff}
    \end{itemize}
  \end{block}
  \begin{block}{Receptor 802.11 CSMA}
    \begin{itemize}
      \item se o frame recíbese OK $\rightarrow$ retorna \textbf{ACK} após \textbf{SIFS} segundos
    \end{itemize}
  \end{block}
  \begin{exampleblock}{NAV --- Network Allocation Vector}
    \begin{itemize}
      \item o frame 802.11 ten un campo co tempo de transmissão
      \item outros nós (escoltando) difiren o acceso por \textbf{NAV} unidades de tempo
    \end{itemize}
  \end{exampleblock}
\end{frame}

\begin{frame}{Problema do terminal oculto}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item gran diferenza na forza do sinal, problemas do canal físico (\emph{shadowing})
        \item dúas estações não podem comunicarse na mesma área
      \end{itemize}
      \textbf{Terminal oculto:}
      \begin{enumerate}
        \item A e C não se ouen
        \item A transmite a B
        \item C quere transmitir a B; C não pode recibir A; C sente o medio desocupado (\emph{carrier sense fails})
        \item colisión em B
        \item A não pode detectar a colisión (\emph{collision detection fails})
        \item A está ``oculto'' para C
      \end{enumerate}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \begin{tikzpicture}[font=\scriptsize,
            n/.style={draw,circle,minimum size=7mm,fill=aneliseAzul!15,inner sep=0.5mm},
          ]
          \node[n](A){A};
          \node[n,right=1.8cm of A](B){B};
          \node[n,right=1.8cm of B](C){C};
          \draw[->,thick,aneliseLaranja] (A) -- node[midway,above=1mm,font=\tiny]{TX} (B);
          \draw[->,thick,aneliseLaranja] (C) -- node[midway,below=1mm,font=\tiny]{TX} (B);
          \draw[dashed,thick,aneliseGris] (A) to[bend left=25] node[midway,below=4mm,font=\tiny]{não se oeñen} (C);
          \node[below=2mm of B,font=\tiny]{colisión em B};
        \end{tikzpicture}
      \end{center}
      \begin{alertblock}{Consecuencia}
        802.11 não usa detección de portadora pura; colisións só dedúcense após a transmissão (CRC / ausencia de ACK).
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Acceso ao canal: IFS e backoff}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{DIFS --- Distributed Inter Frame Space}
      \begin{itemize}
        \item uma estação pode transmitir só se sente o canal desocupado por un tempo \textbf{DIFS}
        \item objetivo: separar datos e ACK com IFS distintos (SIFS $<$ DIFS)
      \end{itemize}
      \textbf{Backoff}
      \begin{itemize}
        \item quando o canal inicialmente está ocupado, a estação deve diferir; quando queda desocupado por DIFS, difire ainda por un tempo aleatorio (\emph{backoff time})
        \item \textbf{slotted backoff}: extraer $K\in(0,W-1)$; decrementar cada slot-time
        \item \textbf{backoff freezing}: o contador ``conxélase'' mentres o canal estea ocupado
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Regras de backoff}
      \begin{itemize}
        \item primeiro: $K\in(0,\mathrm{CW}_{\min})$
        \item sem éxito: $K\in(0, 2(\mathrm{CW}_{\min}+1)-1)$
        \item novamente: $K\in(0, 2^2(\mathrm{CW}_{\min}+1)-1)$
        \item após $m$ fallos: $K\in(0, 2^m(\mathrm{CW}_{\min}+1)-1)$
      \end{itemize}
      \begin{exampleblock}{802.11 clásico}
        $\mathrm{CW}_{\min}=31$, $\mathrm{CW}_{\max}=1023$ ($m=5$). Resultado: \textbf{backoff exponencial}.
      \end{exampleblock}
      \begin{center}
        \begin{tikzpicture}[font=\scriptsize]
          \node[draw,fill=aneliseVerde!12,rounded corners=2mm,text width=5.6cm,align=center](res){Referencia: G.\ Bianchi, ``Performance Analysis of the IEEE 802.11 DCF'', IEEE JSAC 18(3), 2000.};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\section{RTS/CTS}

\begin{frame}{RTS/CTS --- Request-To-Send / Clear-To-Send}
  \begin{itemize}
    \item \textbf{4-way handshaking} no lugar do 2-way básico
    \item introducido por dúas razóns:
      \begin{enumerate}
        \item resolve o problema do \textbf{terminal oculto}
        \item aumenta o throughput com pacotes longos
      \end{enumerate}
  \end{itemize}
  \begin{block}{Mecanismo}
    \begin{itemize}
      \item o transmissor envía \textbf{RTS} curto
      \item o receptor responde \textbf{CTS}
      \item CTS reserva o canal para o transmissor, notificando às outras estações (posiblemente ocultas)
      \item RTS e CTS levan o tempo que o canal estará ocupado $\rightarrow$ as outras actualizan o \textbf{NAV} (\emph{virtual carrier sensing})
    \end{itemize}
  \end{block}
  \begin{exampleblock}{Pros/contras}
    RTS/CTS curtos $\rightarrow$ colisións menos probables e de menor duración. Overhead maior.
  \end{exampleblock}
\end{frame}

\section{PCF e resumo}

\begin{frame}{PCF --- Point Coordination Function}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item mecanismo de acceso baseado em \textbf{turno} (\emph{polling})
        \item o acceso decide o \textbf{point coordinator} (PC): tipicamente o AP
        \item acceso libre de contención (sem colisións no canal)
        \item parte \textbf{opcional} da especificación $\rightarrow$ case nunca implementado; posible uso com 802.11e
        \item os APs envían \textbf{beacons} em intervalos regulares (0,1~s)
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Dúas períodos entre beacons:}
      \begin{itemize}
        \item \textbf{CFP} (Contention Free Period): o AP envía \emph{CF-Poll} a cada estação
        \item \textbf{CP} (Contention Period): úsase DCF
      \end{itemize}
      \begin{alertblock}{Limitacións}
        \begin{itemize}
          \item transición CP$\to$CFP não determinista (espera a fin da transmissão)
          \item round robin interroga todas as estações, mesmo sem datos $\rightarrow$ problemas de escalabilidade
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Resumo dos protocolos MAC}
  \begin{block}{Como se xestiona un canal compartido?}
    \begin{enumerate}
      \item \textbf{Particionamento do canal} (tempo, frequência, código): TDMA, FDMA, CDMA
      \item \textbf{Particionamento aleatorio (dinámico)}: ALOHA, S-ALOHA, CSMA, CSMA/CD
        \begin{itemize}
          \item detección de portadora: fácil em cables, difícil em sem fio
          \item CSMA/CD: Ethernet \quad CSMA/CA: 802.11
        \end{itemize}
      \item \textbf{Pasaxe de turno}: polling centralizado, token
    \end{enumerate}
  \end{block}
  \begin{alertblock}{Evolución (atualizado)}
    802.11e/ax (OFDMA, QoS), 802.11s mesh, 5G NR (scheduling centralizado) son evoluções destas ideas.
  \end{alertblock}
\end{frame}
