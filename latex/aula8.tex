\section{Introdução}

\begin{frame}{IoT --- Internet das Coisas}
  \begin{block}{Definição}
    Rede de objetos físicos (\emph{coisas}) equipados com sensores, atuadores e conectividade, capazes de coletar e trocar dados.
  \end{block}
  \begin{itemize}
    \item Projeção: 30--50 bilhões de dispositivos IoT até 2030
    \item Desafios: bateria, alcance, custo, densidade, segurança
  \end{itemize}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Tecnologias LPWAN}
      \begin{itemize}
        \item LoRaWAN
        \item Sigfox
        \item NB-IoT / LTE-M
        \item Wi-SUN, mioty
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Atributos típicos}
      \begin{itemize}
        \item alcance: 2--15~km (rural até 50~km)
        \item taxa: 0,1--250~kbps
        \item bateria: 5--10 anos
        \item custo de módulo: \$2--\$10
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{IoT e IPv6: 6LoWPAN, RPL e CoAP}
  \begin{columns}

        \begin{itemize}
          \item Endereçamento direto e \emph{end-to-end} de cada sensor
          \item Roteamento padrão, integração com a Internet
          \item 6LoWPAN adapta IPv6 a redes \emph{lossy} de baixa potência (IEEE 802.15.4, \emph{mesh})

        \item \textbf{RPL} (RFC 6550): roteamento para \emph{Low-power and Lossy Networks} (DODAG)
        \item \textbf{CoAP} (RFC 7252): REST sobre UDP
      \end{itemize}

\section{LoRaWAN}

\begin{frame}{LoRaWAN --- arquitetura e operação}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{Camada física: LoRa (CSS)}
      \begin{itemize}
        \item Chirp Spread Spectrum (sub-GHz: 868/915~MHz)
        \item 6 \emph{spreading factors} (SF7--SF12)
        \item sensibilidade até $-137$~dBm
      \end{itemize}
      \textbf{Camada MAC: LoRaWAN}
      \begin{itemize}
        \item topologia \emph{star-of-stars}
        \item \textbf{Classes}: A (ALOHA), B (beacon), C (escuta contínua)
        \item \emph{Adaptive Data Rate} (ADR)
        \item autenticação AES-128 (Join-Request/Accept)
      \end{itemize}
    \end{column}

\end{frame}

\begin{frame}{LoRaWAN: classes e ADR}
  \begin{itemize}
    \item \textbf{Classe A}: uplink seguido de duas janelas de downlink (mínimo consumo)
    \item \textbf{Classe B}: janelas de recepção periódicas (beacon)
    \item \textbf{Classe C}: receptor sempre ligado (downlink contínuo)
  \end{itemize}
  \begin{block}{ADR --- Adaptive Data Rate}
    \begin{itemize}
      \item A rede otimiza SF e potência de transmissão
      \item SF12 $\rightarrow$ maior alcance, menor taxa (290~bps)
      \item SF7 $\rightarrow$ menor alcance, maior taxa (5,4~kbps)
      \item Economia de bateria (até 10 anos com 2 pilhas AA)
    \end{itemize}
  \end{block}
  \note{Enfatizar o trade-off SF: dobrar SF praticamente dobra o ToA; o ADR seleciona o menor SF que garante o link, poupando bateria e ar no canal.}
\end{frame}

\begin{frame}{LoRa: modulação CSS e os \emph{spreading factors}}
  \small
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \textbf{Chirp Spread Spectrum}: cada símbolo é um \emph{chirp} (varredura linear de frequência)
        \item \textbf{SF7--SF12}: \emph{spreading factor} define os \emph{chips} por símbolo; $T_s = 2^{SF}/BW$
        \item SFs distintos são \textbf{ortogonais}: vários fluxos coexistem no mesmo canal com pouca interferência mútua
        \item \emph{Trade-off} fundamental: \textbf{SF maior} $\Rightarrow$ mais alcance, porém menos taxa e mais tempo no ar
      \end{itemize}
      \pause
      \begin{alertblock}{Densidade de rede}
        A ortogonalidade permite ao gateway demodular simultaneamente mensagens em SF7 e SF12 no mesmo canal, multiplicando a capacidade.
      \end{alertblock}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{center}
        {\scriptsize\bfseries Taxa bruta $\times$ SF a BW$=$125~kHz (CR 4/5)}\\[1mm]
        \begin{tabular}{lcc}
          \toprule
          \textbf{SF} & \textbf{Taxa (kbps)} & \textbf{Sensib.\ (dBm)}\\
          \midrule
          SF7 & 5,4 & $-123$\\
          SF8 & 3,1 & $-126$\\
          SF9 & 1,8 & $-129$\\
          SF10 & 0,98 & $-132$\\
          SF11 & 0,44 & $-134$\\
          SF12 & 0,29 & $-137$\\
          \bottomrule
        \end{tabular}\\[1mm]
        {\scriptsize Fonte: LoRa Alliance (valores típicos).}
      \end{center}
      \vspace{0.1cm}
      \begin{block}{Regra prática}
        Dobrar o SF $\approx$ dobra o $T_s$ e o ToA; o ADR escolhe o menor SF que mantém o enlace.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{LoRaWAN: ativação OTAA (Join-Request / Join-Accept)}
  \footnotesize
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{block}{OTAA --- Over-the-Air Activation}
        \begin{enumerate}
          \item \textbf{Join-Request}: o end-device envia \texttt{JoinEUI}, \texttt{DevEUI}, \texttt{DevNonce} (MIC com \texttt{NwkKey})
          \item \textbf{Join-Accept}: o servidor responde com \texttt{AppNonce}, \texttt{NetID}, \texttt{DevAddr} (cifrado com as chaves raiz)
          \item \textbf{Derivação} das chaves de sessão:
            \begin{itemize}
              \item \texttt{NwkSKey} $=$ AES$_{128}$(\texttt{NwkKey},\, AppNonce$|$NetID$|$DevNonce)
              \item \texttt{AppSKey} $=$ AES$_{128}$(\texttt{AppKey},\, AppNonce$|$NetID$|$DevNonce)
            \end{itemize}
        \end{enumerate}
      \end{block}
      \pause
      \begin{alertblock}{ABP --- Activation by Personalization}
        \texttt{NwkSKey}, \texttt{AppSKey} e \texttt{DevAddr} pré-gravados, sem \emph{join}. Mais simples, porém sem rotação de chaves e com risco de clonagem.
      \end{alertblock}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{center}
        \begin{tikzpicture}[font=\scriptsize,
            dev/.style={draw,rounded corners=1.5mm,fill=aneliseAzul!15,inner sep=1.2mm},
            ns/.style={draw,rounded corners=1.5mm,fill=aneliseVerde!20,inner sep=1.2mm},
          ]
          \node[dev](d){End-device};
          \node[ns,right=2.3cm of d](s){\shortstack{Network\\Server}};
          \draw[thick,aneliseLaranja,-{Stealth}] (d) -- (s) node[midway,above]{Join-Request};
          \draw[thick,aneliseVerde,-{Stealth}] (s) -- (d) node[midway,below]{Join-Accept};
          \node[below=1.4cm of s,font=\scriptsize,align=center,text=aneliseGris]{\shortstack{\texttt{NwkSKey} e \texttt{AppSKey}\\(sessão ativa)}};
        \end{tikzpicture}\\[1mm]
        {\scriptsize \textbf{Figura:} fluxo OTAA. Após o aceite, o dispositivo deriva as chaves de sessão para cifrar e autenticar os dados.}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{LoRaWAN: o papel do Network Server}
  \small
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \textbf{Gateways multicanal}: rádios com vários demoduladores (ex.: SX1301) escutam todos os SFs e canais simultaneamente
        \item \textbf{Deduplicação}: a mesma mensagem chega por vários gateways; o servidor guarda uma única cópia (RSSI/SINR do melhor link)
        \item \textbf{Seleção do downlink}: o servidor escolhe o gateway de melhor sinal para responder (sem vínculo fixo dispositivo--gateway)
        \item \textbf{Handover implícito}: ao se deslocar, a rede troca de gateway automaticamente, sem intervenção do nó
        \item \textbf{Roaming}: servidores de operadoras distintas trocam contexto (LoRaWAN Backend Interfaces)
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Camadas de um servidor de rede}
        \begin{itemize}
          \item \textbf{Join Server}: chaves raiz e ativação
          \item \textbf{Network Server}: MAC, ADR, deduplicação e downlink
          \item \textbf{Application Server}: dados e \texttt{AppSKey}
        \end{itemize}
      \end{block}
      \pause
      \begin{exampleblock}{Multioperador}
        Em \emph{roaming}, o sensor é autenticado pelo seu \emph{home} network server; o servidor visitado só retransmite o tráfego.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Sigfox}

\begin{frame}{Sigfox --- UNB (Ultra-Narrow Band)}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Banda ISM sub-GHz (868/915~MHz)
        \item Largura de canal: 100~Hz (ultra-estreita)
        \item Uplink: 140 mensagens/dia, 100~bps
        \item Downlink: 4 mensagens/dia (opcional)
        \item Alcance: 10--50~km (rural)
        \item Modelo de operadora (SIGFOX)
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Comparativo LoRaWAN vs.\ Sigfox}
        \begin{itemize}
          \item LoRaWAN: bidirecional, flexível, redes privadas
          \item Sigfox: ultra-simples, unidirecional, operadora
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Sigfox: arquitetura da rede}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{block}{Elementos}
        \begin{itemize}
          \item \textbf{Endpoints}: sensores transmitem em banda ISM (868/915~MHz) a $\approx$100~bps
          \item \textbf{Base stations} (gateways): receptores ultra-sensíveis espalhados na área de cobertura
          \item \textbf{SIGFOX Cloud}: processamento centralizado (modelo de operadora)
        \end{itemize}
      \end{block}
      \begin{block}{Confiança na entrega}
        Cada uplink é transmitido \textbf{3 vezes} em 3 frequências distintas e em instantes ligeiramente diferentes --- diversidade de frequência e de tempo, sem ACK.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{Downlink condicional}: só após um uplink, na janela aberta pelas retransmissões (1\textsuperscript{a} ou 2\textsuperscript{a})
        \item \textbf{Geolocalização}: posicionamento sem GPS, usando TDoA/Doppler das estações base que recebem a mensagem
        \item \textbf{Limites diários}: 140 uplinks/dia (payload 12~B) e 4 downlinks/dia (payload 8~B)
      \end{itemize}
      \pause
      \begin{alertblock}{Custo do serviço}
        Assinatura por dispositivo/ano; não há rede privada local (no LoRaWAN, o dono opera os gateways).
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{NB-IoT e LTE-M}

\begin{frame}{NB-IoT e LTE-M (3GPP)}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{NB-IoT (Narrowband IoT)}
      \begin{itemize}
        \item 200~kHz, dentro da banda LTE
        \item taxa: até 250~kbps (DL), 20~kbps (UL)
        \item penetração indoor superior
        \item sem mobilidade (fixo/estacionário)
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{LTE-M (eMTC)}
      \begin{itemize}
        \item 1,4~MHz, dentro da banda LTE
        \item taxa: até 1~Mbps
        \item suporta mobilidade e \emph{voice}
        \item ideal para rastreadores, wearables
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \begin{block}{Vantagens sobre LPWAN não-celular}
    \begin{itemize}
      \item QoS garantida (recurso dedicado)
      \item segurança nativa (SIM, AKA)
      \item integração com 5G core (mMTC)
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{NB-IoT: modos de operação e categorias}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{Três modos de implantação (espectro licenciado):}
      \begin{itemize}
        \item \textbf{In-band}: 1 PRB dentro da banda LTE (reuso do equipamento existente)
        \item \textbf{Guard-band}: PRBs nas faixas de guarda entre blocos LTE
        \item \textbf{Standalone}: espectro dedicado (ex.: \emph{refarming} de GSM, 200~kHz)
      \end{itemize}
      \pause
      \begin{block}{O recurso é 1 PRB}
        Cada portadora NB-IoT ocupa \textbf{1 resource block} ($\approx$180~kHz), o que dá alta penetração \emph{indoor} ($+20$~dB de margem sobre LTE) e coexistência com o tráfego LTE.
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Categorias (UE)}
        \begin{itemize}
          \item \textbf{NB1} (Rel-13): DL até $\approx$135~kbps, UL $\approx$20~kbps, sem mobilidade
          \item \textbf{NB2} (Rel-14): DL até $\approx$250~kbps; mais canais, TDD e melhor latência
        \end{itemize}
      \end{block}
      \begin{alertblock}{Nota}
        Valores de \emph{pico teórico} com repetição mínima; na prática a taxa depende da cobertura e das repetições (\emph{coverage enhancement}).
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{LTE-M (Cat-M1): a banda larga da LPWAN celular}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Largura de banda: \textbf{1,4~MHz} (6 PRBs) dentro da banda LTE
        \item Taxa: até $\approx$1~Mbps (DL/UL) --- maior que a do NB-IoT
        \item \textbf{Mobilidade}: suporta \emph{handover}, habilitando rastreadores e \emph{fleet tracking}
        \item \textbf{Voz}: \emph{Voice over LTE} (VoLTE) para dispositivos com fala (wearables, alarmes)
        \item \emph{Half-duplex} opcional para reduzir o custo do transceptor
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Comparando com NB-IoT}
        \begin{itemize}
          \item LTE-M: mais taxa, mobilidade e voz; maior consumo
          \item NB-IoT: menor banda, maior alcance \emph{indoor} e custo menor
        \end{itemize}
      \end{block}
      \pause
      \begin{exampleblock}{Escolha de mercado}
        \textbf{NB-IoT} $\rightarrow$ medidores, equipamentos estacionários, máxima cobertura \emph{indoor}. \textbf{LTE-M} $\rightarrow$ wearables, voz e dispositivos móveis.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Economia de energia: PSM e eDRX}
  \small
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{block}{PSM --- Power Saving Mode}
        Após a última comunicação, o UE negocia um período em que \textbf{não escuta paginamento}, acordando só para transmitir ou fazer a TAU. Consumo cai para $\mu$A --- baterias de 10+ anos.
      \end{block}
      \begin{block}{eDRX --- extended DRX}
        O UE escuta paginamento em \emph{janelas} separadas por intervalos estendidos: LTE-M até $\approx$40,96~s; NB-IoT até $\approx$2,9~h. Compromisso entre latência e consumo.
      \end{block}
      \pause
      \begin{alertblock}{Trade-off}
        \textbf{PSM}: downlink só quando o UE acorda. \textbf{eDRX}: downlink possível, com atraso de até o ciclo. Para aplicações \emph{report-only}, PSM é ideal.
      \end{alertblock}
    \end{column}

\section{Protocolos de aplicação}

\begin{frame}{Protocolos de aplicação para IoT}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{MQTT}
      \begin{itemize}
        \item \emph{Subscribe/Publish} via broker
        \item leve, suporta QoS 0/1/2
        \item padrão de facto para IoT
      \end{itemize}
      \textbf{CoAP}
      \begin{itemize}
        \item UDP + REST (como HTTP)
        \item multicast nativo
        \item muito usado em Constrained RESTful Environments
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{LwM2M (Lightweight M2M)}
      \begin{itemize}
        \item gerenciamento de dispositivo
        \item sobre CoAP ou MQTT
        \item padrão OMA SpecWorks
      \end{itemize}
      \textbf{HTTP/2 e gRPC}
      \begin{itemize}
        \item para gateways/pontos de agregação
        \item streaming bidirecional
      \end{itemize}
    \end{column}
  \end{columns}
  \note{MQTT usa TCP com broker (publish/subscribe); CoAP é UDP/REST com multicast. LwM2M adiciona gerenciamento de ciclo de vida sobre esses protocolos.}
\end{frame}

\begin{frame}{MQTT: publish/subscribe via broker}
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item \textbf{Broker} central (ex.: Mosquitto, EMQX) desacopla produtores e consumidores
        \item \textbf{Tópicos hierárquicos}: \texttt{casa/sala/temperatura}; \emph{wildcards} \texttt{+} (um nível) e \texttt{\#} (subárvore)
        \item \textbf{QoS}: 0 (no máximo uma vez), 1 (pelo menos uma vez, ACK), 2 (exatamente uma vez, \emph{handshake} em 4 fases)
        \item \textbf{Retained}: a última mensagem do tópico fica guardada para novos assinantes
        \item \textbf{Will (LWT)}: o broker publica a ``última vontade'' se o cliente cair sem \texttt{DISCONNECT}
      \end{itemize}

\begin{frame}{CoAP: REST sobre UDP para dispositivos restritos}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \textbf{Modelo REST} (RFC 7252): \texttt{GET}, \texttt{POST}, \texttt{PUT}, \texttt{DELETE} em recursos identificados por URI; porta 5683
        \item \textbf{Confirmável (CON)}: retransmissão com temporizador exponencial e ACK; \textbf{não-confirmável (NON)}: sem ACK (telemetria)
        \item \textbf{Observe} (RFC 7641): assinatura de recurso --- o servidor notifica mudanças (pub/sub sobre REST)
        \item \textbf{Block-wise} (RFC 7959): transferência de payloads grandes em blocos (MTU 1280~B vs.\ 127~B do 802.15.4)
        \item \textbf{Multicast} nativo para descoberta de recursos
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Segurança}
        \begin{itemize}
          \item \textbf{CoAPs}: DTLS sobre UDP (porta 5684), com PSK ou certificados
          \item Contraste com MQTT, que usa TLS sobre TCP
        \end{itemize}
      \end{block}
      \pause
      \begin{exampleblock}{CoAP $\times$ HTTP}
        CoAP é o ``HTTP das redes de sensores'': mesmo modelo de recursos, porém com cabeçalhos comprimidos e controle de congestionamento leve sobre UDP.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{MQTT $\times$ CoAP: escolha do protocolo de aplicação}
  \begin{center}
    \scriptsize
    \setlength{\tabcolsep}{4pt}
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{p{3.2cm}p{4.6cm}p{4.6cm}}
      \toprule
      \textbf{Característica} & \textbf{MQTT (v5.0)} & \textbf{CoAP (RFC 7252)}\\
      \midrule
      Transporte & TCP (orientado a conexão) & UDP (datagrama)\\
      Modelo & publish/subscribe (broker) & request/response (cliente-servidor)\\
      Comunicação & muitos-para-muitos & um-para-um (com multicast)\\
      Qualidade de serviço & QoS 0/1/2 & CON / NON (+ retransmissão)\\
      Segurança & TLS (porta 8883) & DTLS (porta 5684)\\
      Padronização & OASIS (2019) & IETF\\
      Uso típico & dashboards, integrações & dispositivos restritos, multicast\\
      \bottomrule
    \end{tabular}
  \end{center}
  \pause
  \begin{alertblock}{Regra de bolso}
    Broker e aplicações $\rightarrow$ MQTT. Restrições de memória/energia e multicast $\rightarrow$ CoAP. Gateways frequentemente traduzem CoAP$\rightarrow$MQTT.
  \end{alertblock}
\end{frame}

\begin{frame}{Exercício resolvido: tempo no ar de um pacote LoRaWAN}
  \textbf{Problema:} um \emph{end-device} LoRaWAN transmite com \textbf{SF$=12$} e \textbf{BW$=125$~kHz}. O payload tem aproximadamente \textbf{20 símbolos} e o preâmbulo usa 8 símbolos. Estime o tempo no ar (ToA).
  \begin{block}{Resolução}
    \begin{enumerate}
      \item Duração do símbolo:
        \[
          T_s = \frac{2^{SF}}{BW} = \frac{2^{12}}{125\,000} = \frac{4096}{125\,000} \approx 32,8~\text{ms}.
        \]
      \item Número total de símbolos: $n_{\text{preamble}} + n_{\text{payload}} = 8 + 20 = 28$.
      \item Tempo no ar:
        \[
          \text{ToA} = (n_{\text{preamble}} + n_{\text{payload}}) \cdot T_s = 28 \times 32,8~\text{ms} \approx \mathbf{918~\text{ms}}.
        \]
    \end{enumerate}
  \end{block}
  \begin{alertblock}{Leitura do resultado}
    $\approx 0,92$~s por pacote: com o \emph{duty-cycle} de 1\% da banda 868~MHz, o dispositivo deve ficar em silêncio por $\approx 91$~s entre transmissões --- por isso SF12 é reservado a mensagens curtas e raras.
  \end{alertblock}
\end{frame}

\begin{frame}{Resumo da aula}
  \begin{itemize}
    \item \textbf{LPWAN}: baixa taxa, alto alcance e longa duração de bateria para IoT massivo.
    \item \textbf{LoRaWAN}: CSS (868/915~MHz), \emph{star-of-stars}, classes A/B/C e ADR (SF7--SF12).
    \item \textbf{Sigfox}: UNB de 100~Hz, modelo de operadora, limite de 140 uplinks/dia.
    \item \textbf{NB-IoT/LTE-M}: celular, QoS garantida e segurança SIM/AKA; NB-IoT em 200~kHz, LTE-M em 1,4~MHz.
    \item \textbf{Aplicação}: MQTT (broker), CoAP (UDP/REST), LwM2M e gRPC para agregação.
  \end{itemize}
\end{frame}

\begin{frame}{Autoavaliação}
  \begin{enumerate}
    \item Na classe A do LoRaWAN, o downlink é recebido:
      \begin{itemize}
        \item (a) continuamente, com o receptor sempre ligado
        \item (b) em duas janelas abertas logo após o uplink
        \item (c) apenas em janelas periódicas sincronizadas por \emph{beacon}
      \end{itemize}
      \pause
      \textbf{Gabarito:} (b)
    \item Com relação ao ADR, um SF maior (ex.: SF12):
      \begin{itemize}
        \item (a) aumenta a taxa de dados e o alcance
        \item (b) aumenta o alcance, mas reduz a taxa e aumenta o ToA
        \item (c) reduz o consumo de energia sem afetar o alcance
      \end{itemize}
      \pause
      \textbf{Gabarito:} (b)
    \item Comparando LoRaWAN com NB-IoT, uma vantagem do NB-IoT é:
      \begin{itemize}
        \item (a) operar em espectro ISM não licenciado
        \item (b) QoS garantida e segurança nativa (SIM/AKA)
        \item (c) não depender de operadora celular
      \end{itemize}
      \pause
      \textbf{Gabarito:} (b)
    \item MQTT e CoAP se diferenciam principalmente porque:
      \begin{itemize}
        \item (a) MQTT usa \emph{publish/subscribe} via broker e CoAP é REST sobre UDP
        \item (b) ambos exigem TCP obrigatoriamente
        \item (c) CoAP não oferece multicast
      \end{itemize}
      \pause
      \textbf{Gabarito:} (a)
  \end{enumerate}
\end{frame}

\begin{frame}{Desafios de implantação LPWAN}
  \footnotesize
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{Energia}
        \begin{itemize}
          \item Meta de 5--10 anos com bateria
          \item Downlink/ACKs custam caro $\rightarrow$ minimizá-los
          \item ADR, classe A e \emph{duty-cycle} como alavancas
        \end{itemize}
      \end{block}
      \begin{block}{Segurança}
        \begin{itemize}
          \item ISM aberto $\rightarrow$ \emph{jamming} e \emph{spoofing}
          \item AES-128 (LoRaWAN); SIM/AKA (3GPP)
          \item Provisionamento e rotação de chaves (OTAA)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Escala e densidade}
        \begin{itemize}
          \item Milhões de nós $\rightarrow$ deduplicação e escalabilidade
          \item Colisões no ISM (ALOHA-like) limitam a densidade
          \item SFs ortogonais + canais múltiplos aumentam capacidade
        \end{itemize}
      \end{block}
      \begin{block}{Interoperabilidade}
        \begin{itemize}
          \item LoRaWAN 1.0.4 vs.\ versões de fabricante
          \item Roaming entre operadoras; \emph{lock-in} (Sigfox)
          \item Padronização 3GPP (NB-IoT) garante interop.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

\section{Exercícios e quiz}

\begin{frame}{Exercício resolvido: estimativa de vida de bateria}
  \footnotesize
  \textbf{Problema:} sensor LoRaWAN com bateria de \textbf{2400~mAh}. A cada transmissão consome \textbf{100~mA} por \textbf{0,918~s} (ToA do SF12) e envia a cada \textbf{15~min}; em repouso consome \textbf{10~$\mu$A}. Estime a duração da bateria.
  \pause
  \begin{block}{Hipóteses}
    \begin{itemize}
      \item Ciclo de trabalho: $t_{\text{tx}}/\Delta T = 0{,}918/900 \approx 0{,}10\%$
      \item Desprezam-se conversão, envelhecimento, temperatura e auto-descarga
      \item TX de 100~mA $=$ 100\,000~$\mu$A
    \end{itemize}
  \end{block}
  \pause
  \begin{block}{Cálculo da corrente média}
    \[
      I_{\text{média}} = 10 + \frac{0{,}918}{900}\cdot 100\,000 = 10 + 102 \approx \mathbf{112~\mu A}.
    \]
    \[
      \text{Vida} = \frac{2400~\text{mAh}}{0{,}112~\text{mA}} \approx 21\,400~\text{h} \approx \mathbf{2{,}4~\text{anos}}.
    \]
  \end{block}
  \pause
  \begin{alertblock}{Discussão}
    Intervalo de 5~min ou SF7 (ToA $\approx$ 29~ms) muda drasticamente o consumo. \emph{Exercício:} recalcular para SF7 e comparar com os 2,4 anos do SF12.
  \end{alertblock}
\end{frame}
